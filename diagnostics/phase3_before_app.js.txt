  if (window.pdfjsLib) {
    window.pdfjsLib.GlobalWorkerOptions.workerSrc =
      'https://cdn.jsdelivr.net/npm/pdfjs-dist@3.11.174/build/pdf.worker.min.js';
  }
}

/**
 * Synchronizes the rate select dropdown reliably with floating-point tolerance,
 * ensuring visible text is never lost or blank in any theme or browser.
 * @param {number|string} rate
 */
export function syncTtsRateSelect(rate) {
  if (!DOM.ttsRateSelect) return;
  const numRate = parseFloat(rate) || 1.0;

  let matched = false;
  for (let i = 0; i < DOM.ttsRateSelect.options.length; i++) {
    const opt = DOM.ttsRateSelect.options[i];
    if (Math.abs(parseFloat(opt.value) - numRate) < 0.04) {
      DOM.ttsRateSelect.selectedIndex = i;
      matched = true;
      break;
    }
  }

  // If rate doesn't match standard options (e.g. customized via hotkeys), append option
  if (!matched) {
    const label = `${numRate.toFixed(numRate % 1 === 0 ? 1 : 2)}x`;
    const newOpt = new Option(label, numRate.toString(), true, true);
    DOM.ttsRateSelect.add(newOpt);
    DOM.ttsRateSelect.selectedIndex = DOM.ttsRateSelect.options.length - 1;
  }
}

/**
 * Initializes Dexie storage and loads persistent user preferences.
 */
async function initStorageAndSettings() {
  try {
    await initDB();

    // Load persisted settings
    const savedTheme = await getSetting('theme', 'theme-light');
    const savedFont = await getSetting('fontFamily', "'Merriweather', serif");
    const savedFontSize = await getSetting('fontSize', 19);
    const savedLineHeight = await getSetting('lineHeight', 1.8);
    const savedMaxWidth = await getSetting('maxWidth', 820);
    const savedTextAlign = await getSetting('textAlign', 'justify');
    const savedTtsRate = await getSetting('ttsRate', 1.0);
    const savedTtsPitch = await getSetting('ttsPitch', 1.0);
    const savedTtsAutoScroll = await getSetting('ttsAutoScroll', true);
    const savedTtsVoice = await getSetting('ttsVoice', '');
    const savedTtsEngine = await getSetting('ttsEngine', 'browser');
    const savedTtsCloudProvider = await getSetting('ttsCloudProvider', 'fpt');
    const savedTtsCloudApiKey = await getSetting('ttsCloudApiKey', '');
    const savedTtsCloudEndpoint = await getSetting('ttsCloudEndpoint', '');
    const savedTtsCloudVoice = await getSetting('ttsCloudVoice', 'banmai');

    state.settings = {
      theme: savedTheme,
      fontFamily: savedFont,
      fontSize: Number(savedFontSize),
      lineHeight: Number(savedLineHeight),
      maxWidth: Number(savedMaxWidth),
      textAlign: savedTextAlign,
      ttsRate: Number(savedTtsRate),
      ttsPitch: Number(savedTtsPitch),
      ttsVoice: savedTtsVoice,
      ttsEngine: savedTtsEngine,
      ttsCloudProvider: savedTtsCloudProvider,
      ttsCloudApiKey: savedTtsCloudApiKey,
      ttsCloudEndpoint: savedTtsCloudEndpoint,
      ttsCloudVoice: savedTtsCloudVoice,
      ttsAutoScroll: !!savedTtsAutoScroll
    };

    applyTheme(state.settings.theme);
    applyTypography();

    // Sync settings UI controls
    DOM.sliderFontSize.value = state.settings.fontSize;
    DOM.labelFontSize.textContent = `${state.settings.fontSize}px`;
    DOM.sliderLineHeight.value = state.settings.lineHeight;
    DOM.labelLineHeight.textContent = state.settings.lineHeight;
    DOM.sliderMaxWidth.value = state.settings.maxWidth;
    DOM.labelMaxWidth.textContent = `${state.settings.maxWidth}px`;
    syncTtsRateSelect(state.settings.ttsRate);
    if (DOM.btnDockAutoscroll) {
      DOM.btnDockAutoscroll.classList.toggle('active', state.settings.ttsAutoScroll);
    }

    const savedCleaner = await getSetting('enableGlobalCleaner', true);
    state.enableGlobalCleaner = savedCleaner !== false;
    if (DOM.chkGlobalCleaner) {
      DOM.chkGlobalCleaner.checked = state.enableGlobalCleaner;
    }

  } catch (err) {
    console.error('Không thể khởi tạo cơ sở dữ liệu IndexedDB:', err);
    showToast('Lỗi nạp cơ sở dữ liệu: ' + err.message, 'error');
  }
}

let isChapterSwitching = false;
let chapterSwitchHistory = [];
let ttsProgressSaveTimeout = null;

function flushTtsProgressSave() {
  if (ttsProgressSaveTimeout) {
    clearTimeout(ttsProgressSaveTimeout);
    ttsProgressSaveTimeout = null;
    if (state.currentBook) {
      updateProgress(state.currentBook.id, state.currentChapterIndex, calculateOverallProgress(), window.scrollY, state.currentBook.currentChunkIndex || 0);
    }
  }
}

/**
 * Initializes Text-to-Speech Engine with listeners.
 */
function initTTSEngine() {
  state.tts = new TTSEngine();
  state.tts.setRate(state.settings.ttsRate);
  state.tts.setPitch(state.settings.ttsPitch);
  state.tts.setAutoScroll(state.settings.ttsAutoScroll);
  state.tts.setEngine(state.settings.ttsEngine);
  if (state.settings.ttsVoice && state.settings.ttsEngine === 'browser') {
    state.tts.setVoice(state.settings.ttsVoice);
  }
  state.tts.setCloudConfig({
    provider: state.settings.ttsCloudProvider,
    apiKey: state.settings.ttsCloudApiKey,
    endpoint: state.settings.ttsCloudEndpoint,
    voice: state.settings.ttsCloudVoice
  });

  // Actual playback edges only (utterance start/audio playing), separate from intent/status UI.
  state.tts.onPlaybackStateChange = (isPlaying, timestamp) => {
    actualTtsPlaybackActive = isPlaying;
    const signal = isPlaying ? markReadingHistoryTtsStart : markReadingHistoryTtsPause;
    void signal(timestamp).catch(error => console.warn('Reading History TTS signal failed:', error));
  };
  state.tts.onActualChunkStart = index => {
    const chunk = logicalProgressMap?.chunks.get(Number(index));
    if (!chunk) return;
    const followsPrevious = lastCompletedProgressTtsChunkIndex !== null
      && Number(index) === lastCompletedProgressTtsChunkIndex + 1;
    if (!followsPrevious) {
      recordReadingHistoryChapterProgress({
        chapterIndex: logicalProgressMap.chapterIndex,
        position: chunk.startWord,
        totalWords: logicalProgressMap.totalWords,
        explicitJump: true
      });
    }
    activeProgressTtsChunkIndex = Number(index);
  };
  state.tts.onActualChunkComplete = index => {
    const chunk = logicalProgressMap?.chunks.get(Number(index));
    if (!chunk || activeProgressTtsChunkIndex !== Number(index)) return;
    recordReadingHistoryChapterProgress({
      chapterIndex: logicalProgressMap.chapterIndex,
      position: chunk.endWord,
      totalWords: logicalProgressMap.totalWords
    });
    lastCompletedProgressTtsChunkIndex = Number(index);
    activeProgressTtsChunkIndex = null;
  };
  state.tts.onNaturalComplete = () => {
    const targetChapter = Number.isSafeInteger(currentTtsChapterIndex) ? currentTtsChapterIndex : state.currentChapterIndex;
    if (targetChapter >= 0) {
      markReadingHistoryChapterCompleted(targetChapter, Date.now());
    }
  };

  // When voices are loaded or updated
  state.tts.onVoicesLoaded = (voices, selectedVoice) => {
    populateVoiceSelect(voices, selectedVoice);
  };

  // Warning when cloud audio encounters error and falls back to Web Speech
  state.tts.onFallbackWarning = (msg) => {
    showToast(msg, 'warning');
  };

  // When TTS speaks a chunk
  state.tts.onChunkStart = (index, text, total) => {
    // Bước 8: Chỉ gỡ khóa an toàn sau khi câu đầu tiên của chương mới đã cất tiếng đọc thành công
    if (isChapterSwitching) {
      isChapterSwitching = false;
      console.log('>>> [VietReader] Đã cất tiếng đọc câu đầu tiên chương mới. Đã mở khóa an toàn isChapterSwitching.');
    }

    highlightSentenceChunk(index);
    if (DOM.ttsPreviewText) {
      DOM.ttsPreviewText.textContent = text;
    }
    const percent = total > 0 ? Math.round(((index + 1) / total) * 100) : 0;
    if (DOM.dockProgressLabel) {
      DOM.dockProgressLabel.textContent = `Chương ${state.currentChapterIndex + 1} · Đoạn ${index + 1}/${total} (${percent}%)`;
    }
    if (DOM.dockScrubber) {
      DOM.dockScrubber.value = index;
    }
    if (DOM.miniFabBadge) {
      DOM.miniFabBadge.textContent = `${percent}%`;
    }
    if (state.sleepTimer) {
      state.sleepTimer.notifySentenceProgress(index, total);
    }

    if (state.currentBook) {
      state.currentBook.currentChunkIndex = index;
      clearTimeout(ttsProgressSaveTimeout);
      ttsProgressSaveTimeout = setTimeout(() => {
        ttsProgressSaveTimeout = null;
        if (state.currentBook) {
          updateProgress(state.currentBook.id, state.currentChapterIndex, calculateOverallProgress(), window.scrollY, index);
        }
      }, 500);
    }
  };

  // When chunk ends
  state.tts.onChunkEnd = (index) => {
    unhighlightSentenceChunk(index);
  };

  // TTS playback status changed
  state.tts.onStatusChange = async (status) => {
    updateTTSPlayPauseButton(status);

    if (status === 'paused' || status === 'stopped' || status === 'error' || status === 'completed') {
      flushTtsProgressSave();
    }

    // Media Session & Ambient Sound sync
    if (state.mediaSession) {
      if (status === 'playing') {
        state.mediaSession.startBackgroundSession();
        state.ambientMixer?.resumeAll();
      } else if (status === 'paused') {
        state.mediaSession.pauseBackgroundSession();
        state.ambientMixer?.pauseAll();
      } else if (status === 'stopped' || status === 'error') {
        state.mediaSession.stopBackgroundSession();
      } else if (status === 'completed') {
        state.mediaSession.stopBackgroundSession();
        state.sleepTimer?.notifyChapterEnd();
      }
    }

    // TỰ ĐỘNG CHUYỂN CHƯƠNG & ĐỌC TIẾP
    if (status === 'completed') {
      // 1. Khóa nguyên tử (Atomic Mutex Lock)
      if (isChapterSwitching) {
        console.warn('>>> [VietReader] Đang trong tiến trình chuyển chương (isChapterSwitching = true). Bỏ qua sự kiện lặp.');
        return;
      }

      // 2. Cầu chì chống lặp vô hạn (Rate Limiter / Circuit Breaker)
      // Giới hạn: Trong vòng 5 giây không được phép kích hoạt chuyển chương quá 2 lần
      const now = Date.now();
      chapterSwitchHistory = chapterSwitchHistory.filter(t => now - t < 5000);
      if (chapterSwitchHistory.length >= 2) {
        console.error('>>> [VietReader] Circuit Breaker: Kích hoạt ngắt khẩn cấp do chuyển chương quá 2 lần trong 5 giây!');
        if (state.tts) state.tts.stop();
        isChapterSwitching = false;
        showToast('Phát hiện lỗi lặp chuyển chương. Tự động ngắt để bảo vệ bộ nhớ máy tính.', 'error');
        return;
      }
      chapterSwitchHistory.push(now);

      // Kích hoạt khóa nguyên tử trong suốt quá trình chuyển chương
      isChapterSwitching = true;
      console.log('>>> [VietReader] Hoàn thành chương, bắt đầu luồng chuyển chương an toàn tuần tự...');

      // Kiểm tra hẹn giờ "Dừng khi hết chương"
      if (state.sleepTimer?.isStopAtChapterEnd && state.sleepTimer.isStopAtChapterEnd()) {
        state.sleepTimer?.consumeChapterEndTrigger?.();
        isChapterSwitching = false;
        showToast('Đã dừng theo hẹn giờ tắt khi hết chương.', 'info');
        return;
      }

      // Kiểm tra nếu còn chương tiếp theo
      const totalChapters = state.chaptersMeta?.length
        || state.currentBook?.chapters?.length
        || state.chapters?.length
        || state.totalChapters
        || 0;

      if (state.currentChapterIndex < totalChapters - 1) {
        try {
          // Bước 1: Dừng toàn bộ âm thanh, xóa sạch currentUtterance và timers
          if (state.tts) {
            state.tts.stop();
          }

          // Bước 2: Hiện Toast thông báo
          showToast('Đã đọc hết chương. Chuyển chương tiếp theo...', 'info');

          // Bước 3: Gọi hàm chuyển chương bất đồng bộ
          const nextIndex = state.currentChapterIndex + 1;
          await loadChapter(nextIndex);

          // Bước 4 - Nghỉ bắt buộc: Chờ khoảng đệm an toàn 800ms - 1000ms để trình duyệt hoàn tất render DOM và Garbage Collector dọn sạch RAM
          await new Promise(resolve => setTimeout(resolve, 800));

          // Bước 5 - Kiểm tra dữ liệu hợp lệ: Lấy nội dung chương mới, kiểm tra nghiêm ngặt
          const newContent = state.currentChapter?.content || '';
          if (!newContent || newContent.trim().length < 20) {
            console.warn('>>> [VietReader] Nội dung chương mới rỗng hoặc dưới 20 ký tự. Dừng phát để tránh vòng lặp 0ms.');
            if (state.tts) state.tts.stop();
            isChapterSwitching = false;
            showToast('Chương mới không có đủ nội dung văn bản.', 'warning');
            return;
          }

          // Bước 6: Nạp nội dung mới vào state.tts.loadText
          const rules = await getCustomRules(state.currentBook.id);
          const cleanedNewContent = cleanChapterText(newContent, {
            rules,
            enableCleaner: state.enableGlobalCleaner
          });
          state.tts.loadText(cleanedNewContent || newContent);

          // Bước 7: Bắt đầu đọc từ câu đầu tiên
          if (state.tts.chunks && state.tts.chunks.length > 0) {
            console.log('>>> [VietReader] Khởi động phát TTS cho chương mới từ câu đầu tiên...');
            state.tts.jumpToIndex(0, true);

            // Timeout an toàn phòng hờ nếu trình duyệt không bắn onChunkStart do lỗi engine
            setTimeout(() => {
              if (isChapterSwitching) {
                console.warn('>>> [VietReader] Timeout an toàn: Tự động giải phóng isChapterSwitching.');
                isChapterSwitching = false;
              }
            }, 8000);
          } else {
            console.warn('>>> [VietReader] Mảng chunks rỗng sau khi nạp chương mới.');
            if (state.tts) state.tts.stop();
            isChapterSwitching = false;
          }

        } catch (err) {
          console.error('>>> [VietReader] Lỗi khi tự động chuyển chương:', err);
          if (state.tts) state.tts.stop();
          isChapterSwitching = false;
          showToast('Lỗi khi tải chương kế tiếp.', 'error');
        }
      } else {
        isChapterSwitching = false;
        showToast('🎉 Bạn đã nghe hết toàn bộ tác phẩm!', 'success');
      }
    }
  };
  populateVoiceSelect(state.tts.voices, state.tts.selectedVoice);
}

/**
 * Populates voice options in the dropdown with categorized groups:
 * 1. Cloud AI voices (FPT AI & OpenAI)
 * 2. Web Speech System & Neural Voices (HoaiMy, Nam, Google...)
 * 3. Other languages
 */
function populateVoiceSelect(voices, activeVoice) {
  if (!DOM.ttsVoiceSelect) return;
  DOM.ttsVoiceSelect.innerHTML = '';

  // 1. Group: Cloud AI Voices
  const optgroupCloud = document.createElement('optgroup');
  optgroupCloud.label = '⚡ Giọng Cloud AI (FPT AI & OpenAI)';
  VIETNAMESE_VOICE_CATALOG.cloud.forEach(v => {
    const opt = document.createElement('option');
    opt.value = v.id;
    opt.textContent = v.name;
    optgroupCloud.appendChild(opt);
  });
  DOM.ttsVoiceSelect.appendChild(optgroupCloud);

  // Filter Vietnamese browser voices
  const viVoices = [];
  const otherVoices = [];

  voices.forEach(v => {
    const isVi = v.lang && (v.lang.toLowerCase() === 'vi-vn' || v.lang.toLowerCase().startsWith('vi'));
    if (isVi) viVoices.push(v);
    else otherVoices.push(v);
  });

  // Sort Vietnamese voices: Natural / Online first
  viVoices.sort((a, b) => {
    const aPri = /natural|hoaimy|namminh|microsoft nam|google/i.test(a.name) ? -1 : 1;
    const bPri = /natural|hoaimy|namminh|microsoft nam|google/i.test(b.name) ? -1 : 1;
    return aPri - bPri;
  });

  // 2. Group: Browser Web Speech API voices
  if (viVoices.length > 0) {
    const optgroupVi = document.createElement('optgroup');
    optgroupVi.label = '🌐 Giọng Trình Duyệt (Web Speech API)';
    viVoices.forEach(v => {
      const opt = document.createElement('option');
      opt.value = v.voiceURI;
      opt.textContent = formatBrowserVoiceLabel(v);
      optgroupVi.appendChild(opt);
    });
    DOM.ttsVoiceSelect.appendChild(optgroupVi);
  }

  // 3. Group: Other languages
  if (otherVoices.length > 0) {
    const optgroupOther = document.createElement('optgroup');
    optgroupOther.label = 'Giọng ngôn ngữ khác';
    otherVoices.slice(0, 10).forEach(v => {
      const opt = document.createElement('option');
      opt.value = v.voiceURI;
      opt.textContent = `${v.name} (${v.lang})`;
      optgroupOther.appendChild(opt);
    });
    DOM.ttsVoiceSelect.appendChild(optgroupOther);
  }

  // Refreshing the list is a UI operation; only explicit configuration paths change playback.
  const cloudVoice = state.tts.engine === 'cloud'
    ? state.tts.cloudConfig.provider + ':' + state.tts.cloudConfig.voice : '';
  const preferredVoice = cloudVoice || (activeVoice ? activeVoice.voiceURI : '') || state.settings.ttsVoice;
  if (preferredVoice) DOM.ttsVoiceSelect.value = preferredVoice;
}

/**
 * Updates TTS Play/Pause toggle button icon and mini widget state.
 */
function updateTTSPlayPauseButton(status) {
  const isPlaying = status === 'playing';

  if (DOM.btnTtsPlayToggle) {
    DOM.btnTtsPlayToggle.classList.toggle('playing', isPlaying);
  }
  if (DOM.readerMiniFab) {
    DOM.readerMiniFab.classList.toggle('playing', isPlaying);
  }

  if (isPlaying) {
    // Show Pause Icon
    DOM.ttsPlayIcon.innerHTML = `
      <rect x="6" y="4" width="4" height="16"/>
      <rect x="14" y="4" width="4" height="16"/>
    `;
    DOM.btnToggleTtsDock.classList.add('active');
  } else {
    // Show Play Icon
    DOM.ttsPlayIcon.innerHTML = `
      <polygon points="5 3 19 12 5 21 5 3"/>
    `;
    if (status === 'stopped' || status === 'completed') {
      DOM.btnToggleTtsDock.classList.remove('active');
    }
  }
}

/**
 * Highlights a sentence span in the reader view and auto-scrolls to it.
 */
function highlightSentenceChunk(index) {
  document.querySelectorAll('.tts-chunk.active-tts-chunk').forEach(el => {
    el.classList.remove('active-tts-chunk');
  });

  const targetSpan = document.querySelector(`.tts-chunk[data-chunk-index="${index}"]`);
  if (targetSpan) {
    targetSpan.classList.add('active-tts-chunk');
    if (state.settings.ttsAutoScroll) {
      targetSpan.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }
  }
}

/**
 * Clears highlight from a chunk.
 */
function unhighlightSentenceChunk(index) {
  const targetSpan = document.querySelector(`.tts-chunk[data-chunk-index="${index}"]`);
  if (targetSpan) {
    targetSpan.classList.remove('active-tts-chunk');
  }
}

function noteTrustedKeyboardActivity() {
  if (!DOM.readerView?.classList.contains('active')) return;
  const now = Date.now();
  if (now - lastReadingActivitySignalAt < 750) return;
  lastReadingActivitySignalAt = now;
  void markReadingHistoryActivity({ timestamp: now, chapterIndex: state.currentChapterIndex })
    .catch(error => console.warn('Reading History activity signal failed:', error));
}

/**
 * Hotkeys for reading and TTS convenience:
 * - Space: Play / Pause
 * - Left / Right: Previous / Next chunk
 * - Up / Down: Increase / Decrease speed
 * - [ and ] : Previous / Next chapter
 * - Escape : Exit Zen mode
 */
function handleKeyboardShortcuts(e) {
  // Search/bookmark temporary focus and the search modal own keys while focused.
  if (e.defaultPrevented || e.isComposing || DOM.globalSearchModal?.classList.contains('open')
    || (e.key !== 'Escape' && document.activeElement?.closest?.('.search-focus-match, .bookmark-focus'))) return;

  // Let modal/drawer controls own their keys. Escape closes the topmost app surface.
  if (DOM.confirmModal?.classList.contains('open')) return;
  if (e.key === 'Escape') {
    const closableSurfaces = [
      [DOM.ttsSettingsModal, closeTtsSettingsModal],
      [DOM.dictModal, closeDictionaryModal],
      [DOM.sleepTimerModal, closeSleepTimerModal],
      [DOM.ambientMixerModal, closeAmbientMixerModal],
      [DOM.settingsModal, closeSettingsModal],
      [DOM.pasteModal, () => DOM.pasteModal.classList.remove('open')],
      [DOM.backupModal, () => DOM.backupModal.classList.remove('open')],
      [DOM.tocDrawerBackdrop, closeTocDrawer]
    ];
    const openSurface = closableSurfaces.find(([element]) => element?.classList.contains('open'));
    if (openSurface) {
      e.preventDefault();
      e.stopPropagation();
      openSurface[1]();
      return;
    }
  }

  if (DOM.historyView?.classList.contains('active')) {
    if (DOM.confirmModal?.classList.contains('open')) return;
    if (e.key === 'Escape') {
      e.preventDefault();
      closeHistoryView();
    }
    return;
  }

  if (DOM.analyticsView?.classList.contains('active')) {
    if (e.key === 'Escape') {
      e.preventDefault();
      closeAnalyticsView();
    }
    return;
  }

  // Reader shortcuts must not consume typing, editable content, or modal/drawer input.
  const activeElement = document.activeElement;
  const activeTag = activeElement?.tagName?.toLowerCase() || '';
  const isEditable = activeTag === 'input' || activeTag === 'textarea' || activeTag === 'select'
    || activeElement?.isContentEditable
    || !!activeElement?.closest?.('[contenteditable]:not([contenteditable="false"])');
  const appSurfaceOpen = document.querySelector(
    '.modal-overlay.open, .settings-modal-backdrop.open, .drawer-backdrop.open'
  );
  if (isEditable || appSurfaceOpen) return;

  if (DOM.readerView.classList.contains('active')) {
    const readerNavigationKey = e.key === ' ' || e.code === 'Space'
      || ['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown', '[', ']'].includes(e.key)
      || ['BracketLeft', 'BracketRight'].includes(e.code);
    if (readerNavigationKey) noteTrustedKeyboardActivity();
    // Space: Play / Pause TTS
    if (e.key === ' ' || e.code === 'Space') {
      e.preventDefault();
      if (DOM.btnTtsPlayToggle) DOM.btnTtsPlayToggle.click();
      return;
    }

    // Left Arrow: Rewind 1 chunk
    if (e.key === 'ArrowLeft') {
      e.preventDefault();
      if (state.tts) state.tts.prev();
      return;
    }

    // Right Arrow: Forward 1 chunk
    if (e.key === 'ArrowRight') {
      e.preventDefault();
      if (state.tts) state.tts.next();
      return;
    }

    // Up Arrow: Increase speed (+0.1x)
    if (e.key === 'ArrowUp') {
      e.preventDefault();
      changePlaybackRate(0.1);
      return;
    }

    // Down Arrow: Decrease speed (-0.1x)
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      changePlaybackRate(-0.1);
      return;
    }

    // [ : Previous Chapter
    if (e.key === '[' || e.code === 'BracketLeft') {
      e.preventDefault();
      navigateChapter(-1);
      return;
    }

    // ] : Next Chapter
    if (e.key === ']' || e.code === 'BracketRight') {
      e.preventDefault();
      navigateChapter(1);
      return;
    }

    // A or S: Hands-Free Auto-Scroll toggle
    if (e.key === 'a' || e.key === 'A' || e.key === 's' || e.key === 'S') {
      e.preventDefault();
      toggleHandsFreeAutoScroll();
      return;
    }

    // Escape: Exit Zen mode if active or close selection toolbar
    if (e.key === 'Escape') {
      hideSelectionToolbar();
      if (state.isZenMode) {
        e.preventDefault();
        toggleZenMode();
        return;
      }
    }
  }
}

/**
 * Handles batch or single file selection (.txt, .pdf, .epub).
 */
async function handleFileSelection(files) {
  if (!files || files.length === 0) return;

  for (const file of files) {
    const ext = file.name.split('.').pop().toLowerCase();
    showToast(`Đang phân tích file: ${file.name}...`, 'info');

    try {
      let parsed = null;

      if (ext === 'txt') {
        const txtData = await parseTxtFile(file);
        const chapters = splitTextIntoChapters(txtData.rawText, txtData.title);
        parsed = {
          title: txtData.title,
          author: txtData.author,
          chapters,
          wordCount: countWords(txtData.rawText)
        };
      } else if (ext === 'pdf') {
        const pdfData = await parsePdfFile(file, (progress) => {
          if (progress % 25 === 0) showToast(`Đang đọc PDF: ${progress}%...`, 'info');
        });
        const chapters = splitTextIntoChapters(pdfData.rawText, pdfData.title);
        parsed = {
          title: pdfData.title,
          author: pdfData.author,
          chapters,
          wordCount: countWords(pdfData.rawText)
        };
      } else if (ext === 'epub') {
        const epubData = await parseEpubFile(file);
        let chapters = epubData.chapters;
        if (!chapters || chapters.length === 0) {
          chapters = splitTextIntoChapters(epubData.rawText, epubData.title);
        }
        parsed = {
          title: epubData.title,
          author: epubData.author,
          chapters,
          wordCount: countWords(epubData.rawText)
        };
      } else {
        showToast(`Định dạng .${ext} không được hỗ trợ. Vui lòng chọn .txt, .pdf hoặc .epub`, 'error');
        continue;
      }

      if (!parsed || !parsed.chapters || parsed.chapters.length === 0) {
        showToast(`Không bóc tách được nội dung từ file: ${file.name}`, 'error');
        continue;
      }

      const bookId = await saveBookWithChapters(parsed, parsed.chapters);
      showToast(`Đã thêm thành công: "${parsed.title}" (${parsed.chapters.length} chương)!`, 'success');
      await refreshLibrary();

      // Automatically open the first imported book
      await openBook(bookId);

    } catch (err) {
      console.error('Lỗi phân tích file:', err);
      showToast(`Lỗi khi mở ${file.name}: ${err.message}`, 'error');
    }
  }

  DOM.fileInput.value = '';
}

/**
 * Handles submitting raw text from paste modal.
 */
async function handleRawTextSubmit() {
  const title = DOM.pasteTitleInput.value.trim();
  const author = DOM.pasteAuthorInput.value.trim() || 'Khuyết danh';
  const content = DOM.pasteContentInput.value.trim();

  if (!title) {
    showToast('Vui lòng nhập tên tác phẩm', 'warning');
    DOM.pasteTitleInput.focus();
    return;
  }

  if (!content) {
    showToast('Vui lòng dán nội dung truyện vào ô văn bản', 'warning');
    DOM.pasteContentInput.focus();
    return;
  }

  try {
    showToast('Đang phân tách các chương...', 'info');
    const chapters = splitTextIntoChapters(content, title);
    const totalWords = countWords(content);

    const bookId = await saveBookWithChapters({
      title,
      author,
      wordCount: totalWords
    }, chapters);

    DOM.pasteModal.classList.remove('open');
    DOM.pasteTitleInput.value = '';
    DOM.pasteAuthorInput.value = '';
    DOM.pasteContentInput.value = '';

    showToast(`Đã lưu "${title}" với ${chapters.length} chương!`, 'success');
    await refreshLibrary();
    await openBook(bookId);

  } catch (err) {
    console.error('Lỗi lưu văn bản thô:', err);
    showToast('Không thể lưu tác phẩm: ' + err.message, 'error');
  }
}

/**
 * Refreshes the library dashboard view with all books from IndexedDB.
 */
export async function refreshLibrary() {
  try {
    const books = await getAllBooks();
    DOM.booksCount.textContent = books.length;

    if (books.length === 0) {
      DOM.booksGrid.innerHTML = '';
      DOM.emptyLibrary.style.display = 'block';
      return;
    }

    DOM.emptyLibrary.style.display = 'none';
    renderBookCards(books);
  } catch (err) {
    console.error('Lỗi tải danh sách sách:', err);
  }
}

      : 0;
    const progressVal = Math.max(0, Math.min(100, Math.max(chapterPercent, book.scrollProgress || 0)));
    const timeAgoStr = formatTimeAgo(book.lastReadAt);

    return `
      <div class="book-card" data-book-id="${book.id}">
        <div class="book-cover" style="background: ${book.coverGradient};">

            <h3 class="book-cover-title">${escapeHTML(book.title)}</h3>
            <span class="book-cover-author">${escapeHTML(book.author)}</span>
          </div>
        </div>
        <div class="book-card-body">
          <div class="book-meta-stats">
            <span>${book.totalChapters} chương</span>
            <span>${progressVal}% hoàn thành</span>
          </div>
          <div class="book-progress-track">
            <div class="book-progress-bar" style="width: ${progressVal}%;"></div>
          </div>
          <div style="font-size: 0.78rem; color: var(--text-muted); margin-bottom: 0.85rem;">
            Đọc lần cuối: ${timeAgoStr}
          </div>
          <div class="book-card-actions">
            <button class="btn-read" data-action="read" data-book-id="${book.id}">
              <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                <polygon points="5 3 19 12 5 21 5 3"/>
              </svg>
              <span>${book.currentChapterIndex > 0 ? 'Đọc tiếp' : 'Bắt đầu đọc'}</span>
            </button>
            <button class="btn-delete" data-action="delete" data-book-id="${book.id}" title="Xóa sách khỏi kệ">
              <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                <polyline points="3 6 5 6 21 6"/>
                <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"/>
              </svg>
            </button>
          </div>
        </div>
      </div>
    `;
  }).join('');

  // Attach card action listeners
  DOM.booksGrid.querySelectorAll('.book-card').forEach(card => {
    const bookId = card.getAttribute('data-book-id');

    card.querySelector('[data-action="read"]').addEventListener('click', (e) => {
      e.stopPropagation();
      openBook(bookId);
    });

    card.querySelector('[data-action="delete"]').addEventListener('click', (e) => {
      e.stopPropagation();
      handleDeleteBook(bookId);
    });

    card.addEventListener('click', () => {
      openBook(bookId);
    });
  });
}

/**
 * Filter library books by keyword.
 */
function filterLibraryCards(keyword) {
  const query = keyword.trim().toLowerCase();
  const cards = DOM.booksGrid.querySelectorAll('.book-card');
  let visibleCount = 0;

  cards.forEach(card => {
    const title = card.querySelector('.book-cover-title')?.textContent.toLowerCase() || '';
    const author = card.querySelector('.book-cover-author')?.textContent.toLowerCase() || '';
    if (title.includes(query) || author.includes(query)) {
      card.style.display = 'flex';
      visibleCount++;
    } else {
      card.style.display = 'none';
    }
  });

  DOM.booksCount.textContent = visibleCount;
}

function openGlobalSearch() {
  if (DOM.globalSearchModal.classList.contains('open')) return;
  globalSearchReturnFocus = document.activeElement;
  DOM.globalSearchModal.classList.add('open');
  DOM.globalSearchModal.setAttribute('aria-hidden', 'false');
  DOM.globalSearchInput.focus();
  if (DOM.globalSearchInput.value.trim()) {
    handleGlobalSearchInput();
  } else {
    const progress = searchIndexService.getProgress();
    if (progress.indexing && progress.totalChapters > 0) renderGlobalSearchIndexProgress(progress);
    else setGlobalSearchStatus('Nhập ít nhất 2 ký tự để tìm trong nội dung sách.');
  }
}

function invalidateGlobalSearchWork() {
  globalSearchUiGeneration++;
  if (globalSearchTimer) clearTimeout(globalSearchTimer);
  globalSearchTimer = null;
  globalSearchAbortController?.abort();
  globalSearchAbortController = null;
  return globalSearchUiGeneration;
}

function closeGlobalSearch() {
  if (!DOM.globalSearchModal.classList.contains('open')) return;
  invalidateGlobalSearchWork();
  DOM.globalSearchModal.classList.remove('open');
  DOM.globalSearchModal.setAttribute('aria-hidden', 'true');
  DOM.globalSearchResults.setAttribute('aria-busy', 'false');
  DOM.globalSearchResultsList.replaceChildren();
  DOM.btnGlobalSearchMore.hidden = true;
  globalSearchResults = null;
  if (globalSearchReturnFocus?.isConnected) globalSearchReturnFocus.focus();
  globalSearchReturnFocus = null;
}

function setGlobalSearchStatus(message, busy = false, state = busy ? 'searching' : 'idle') {
  DOM.globalSearchStatus.textContent = message;
  DOM.globalSearchResults.setAttribute('aria-busy', busy ? 'true' : 'false');
  DOM.globalSearchModal.dataset.state = state;
  DOM.btnGlobalSearchMore.disabled = busy;
}

function renderGlobalSearchIndexProgress(progress = searchIndexService.getProgress()) {
  if (!DOM.globalSearchModal.classList.contains('open') || DOM.globalSearchInput.value.trim()) return;
  if (progress.indexing && progress.totalChapters > 0) {
    setGlobalSearchStatus(`Đang chuẩn bị chỉ mục tìm kiếm… ${progress.indexedChapters || 0} / ${progress.totalChapters} chương`, true, 'indexing');
  } else {
    setGlobalSearchStatus('Nhập ít nhất 2 ký tự để tìm trong nội dung sách.');
  }
}

function handleGlobalSearchInput() {
  const generation = invalidateGlobalSearchWork();
  globalSearchResults = null;
  globalSearchVisibleCount = 50;
  DOM.globalSearchResultsList.replaceChildren();
  DOM.btnGlobalSearchMore.hidden = true;
  const query = DOM.globalSearchInput.value;
  const normalized = normalizeSearchText(query);
  if (!query.trim()) {
    renderGlobalSearchIndexProgress();
    return;
  }
  if (query.length > MAX_RAW_QUERY_LENGTH || normalized.length > MAX_NORMALIZED_QUERY_LENGTH) {
    setGlobalSearchStatus('Từ khóa quá dài. Hãy rút gọn rồi thử lại.', false, 'error');
    return;
  }
  if (normalized.length < 2) {
    setGlobalSearchStatus('Nhập ít nhất 2 ký tự để bắt đầu tìm kiếm.');
    return;
  }

  setGlobalSearchStatus('Đang chờ tìm…', true);
  globalSearchTimer = setTimeout(() => {
    runGlobalSearch(generation, query, 0, 0);
  }, 250);
}

function clearGlobalSearch() {
  DOM.globalSearchInput.value = '';
  handleGlobalSearchInput();
  DOM.globalSearchInput.focus();
}

async function runGlobalSearch(generation, query, offset = 0, retryCount = 0) {
  if (generation !== globalSearchUiGeneration || !DOM.globalSearchModal.classList.contains('open')) return;
  const controller = new AbortController();
  globalSearchAbortController = controller;
  if (offset === 0) {
    setGlobalSearchStatus('Đang tìm trong thư viện…', true);
  } else {
    DOM.globalSearchResults.setAttribute('aria-busy', 'true');
    DOM.globalSearchStatus.textContent = 'Đang tải thêm kết quả…';
  }

  try {
    const response = await searchIndexService.search(query, {
      limit: DEFAULT_SEARCH_LIMIT,
      offset,
      signal: controller.signal,
      requestId: generation,
      isCurrent: id => id === globalSearchUiGeneration && DOM.globalSearchModal.classList.contains('open'),
      progressCallback: progress => {
        if (generation !== globalSearchUiGeneration || !DOM.globalSearchModal.classList.contains('open')) return;
        const indexing = searchIndexService.getProgress();
        const indexProgress = indexing.indexing && indexing.totalChapters > 0
          ? ` · Chỉ mục ${indexing.indexedChapters || 0}/${indexing.totalChapters}`
          : '';
        DOM.globalSearchStatus.textContent = `Đang tìm… ${progress.indexedChapters}/${progress.totalChapters} chương${indexProgress}`;
      }
    });

    if (generation !== globalSearchUiGeneration || !DOM.globalSearchModal.classList.contains('open')) return;
    if (response.cancelled) {
      if (retryCount < 2) {
        setGlobalSearchStatus('Chỉ mục vừa thay đổi, đang thử lại…', true, 'indexing');
        globalSearchTimer = setTimeout(() => runGlobalSearch(generation, query, offset, retryCount + 1), 150);
      } else {
        setGlobalSearchStatus('Chỉ mục đang thay đổi. Hãy thử tìm lại sau giây lát.', false, 'error');
      }
      return;
    }

    if (offset === 0 || !globalSearchResults) {
      globalSearchResults = response;
      globalSearchVisibleCount = Math.min(50, response.results.length);
    } else {
      globalSearchResults = {
        ...response,
        results: [...globalSearchResults.results, ...response.results]
      };
      globalSearchVisibleCount = Math.min(globalSearchResults.results.length, globalSearchVisibleCount + 50);
    }
    renderGlobalSearchResults();
  } catch {
    if (generation !== globalSearchUiGeneration || controller.signal.aborted) return;
    setGlobalSearchStatus('Không thể tìm kiếm lúc này. Hãy thử lại.', false, 'error');
    DOM.globalSearchResultsList.replaceChildren();
    DOM.btnGlobalSearchMore.hidden = true;
  }
}

function appendSafeHighlightedSnippet(container, snippet) {
  const text = typeof snippet?.text === 'string' ? snippet.text : '';
  const start = Math.max(0, Math.min(text.length, Number(snippet?.matchStart) || 0));
  const end = Math.max(start, Math.min(text.length, Number(snippet?.matchEnd) || start));
  container.append(document.createTextNode(text.slice(0, start)));
  if (end > start) {
    const mark = document.createElement('mark');
    mark.className = 'search-result-match';
    mark.textContent = text.slice(start, end);
    container.append(mark);
  }
  container.append(document.createTextNode(text.slice(end)));
}

function renderGlobalSearchResults() {
  const resultSet = globalSearchResults;
  if (!resultSet) return;
  const matches = resultSet.results.slice(0, globalSearchVisibleCount);
  if (!matches.length) {
    const empty = document.createElement('p');
    empty.className = 'global-search-empty';
    empty.textContent = 'Không tìm thấy nội dung phù hợp.';
    DOM.globalSearchResultsList.replaceChildren(empty);
    setGlobalSearchStatus(`0 kết quả cho “${DOM.globalSearchInput.value.trim()}”.`, false, 'no-results');
    DOM.btnGlobalSearchMore.hidden = true;
    return;
  }

  const groups = [];
  const books = new Map();
  const resultIndices = new Map(resultSet.results.map((match, index) => [match, index]));
  for (const match of matches) {
    const bookKey = String(match.bookId);
    let bookGroup = books.get(bookKey);
    if (!bookGroup) {
      bookGroup = { bookId: bookKey, title: match.bookTitle || 'Tác phẩm', chapters: [], chapterMap: new Map() };
      books.set(bookKey, bookGroup);
      groups.push(bookGroup);
    }
    const chapterKey = String(match.chapterIndex);
    let chapterGroup = bookGroup.chapterMap.get(chapterKey);
    if (!chapterGroup) {
      chapterGroup = { title: match.chapterTitle || `Chương ${Number(match.chapterIndex) + 1}`, matches: [] };
      bookGroup.chapterMap.set(chapterKey, chapterGroup);
      bookGroup.chapters.push(chapterGroup);
    }
    chapterGroup.matches.push({ match, resultIndex: resultIndices.get(match) ?? 0 });
  }

  const fragment = document.createDocumentFragment();
  for (const bookGroup of groups) {
    const bookSection = document.createElement('section');
    bookSection.className = 'global-search-book-group';
    const bookTitle = document.createElement('h3');
    bookTitle.className = 'global-search-book-title';
    bookTitle.textContent = bookGroup.title;
    bookSection.append(bookTitle);
    for (const chapterGroup of bookGroup.chapters) {
      const chapterSection = document.createElement('section');
      chapterSection.className = 'global-search-chapter-group';
      const chapterTitle = document.createElement('h4');
      chapterTitle.className = 'global-search-chapter-title';
      chapterTitle.textContent = chapterGroup.title;
      chapterSection.append(chapterTitle);
      for (const resultEntry of chapterGroup.matches) {
        const { match, resultIndex } = resultEntry;
        const row = document.createElement('article');
        row.className = 'global-search-match-row';
        row.tabIndex = 0;
        row.dataset.searchResult = 'true';
        row.dataset.searchResultIndex = String(resultIndex);
        row.setAttribute('role', 'button');
        row.setAttribute('aria-label', `${bookGroup.title}, ${chapterGroup.title}: ${match.matchedText}`);
        const snippet = document.createElement('span');
        snippet.className = 'global-search-snippet';
        appendSafeHighlightedSnippet(snippet, match.snippet);
        row.append(snippet);
        chapterSection.append(row);
      }
      bookSection.append(chapterSection);
    }
    fragment.append(bookSection);
  }
  DOM.globalSearchResultsList.replaceChildren(fragment);

  const chapterCount = new Set(matches.map(match => `${match.bookId}:${match.chapterIndex}`)).size;
  const bookCount = new Set(matches.map(match => String(match.bookId))).size;
  const loadedCount = resultSet.results.length;
  const isPartial = resultSet.truncated || loadedCount < resultSet.totalMatches;
  const summary = isPartial
    ? `Đang hiển thị ${loadedCount} / ${resultSet.totalMatches} kết quả · ${chapterCount} chương · ${bookCount} truyện`
    : `${resultSet.totalMatches} kết quả · ${chapterCount} chương · ${bookCount} truyện`;
  setGlobalSearchStatus(summary, false, 'results');
  const canLoadMoreLocally = globalSearchVisibleCount < resultSet.results.length;
  const canFetchMore = resultSet.truncated && resultSet.results.length < MAX_SEARCH_LIMIT;
  DOM.btnGlobalSearchMore.hidden = !canLoadMoreLocally && !canFetchMore;
}

  const row = event.target.closest?.('[data-search-result="true"]');
  if (!row || !DOM.globalSearchResultsList.contains(row)) return;
  const resultIndex = Number(row.dataset.searchResultIndex);
  if (Number.isInteger(resultIndex)) void navigateToGlobalSearchResult(resultIndex);
}

function clearSearchFocusHighlight() {
  const parents = new Set();
  for (const node of activeSearchFocusNodes) {
    if (!node?.isConnected || !node.parentNode) continue;
    const parent = node.parentNode;
    while (node.firstChild) parent.insertBefore(node.firstChild, node);
    parent.removeChild(node);
    parents.add(parent);
  }
  for (const parent of parents) parent.normalize?.();
  activeSearchFocusNodes = [];
}

function findSearchAnchorInReader(result) {
  return findSearchAnchor(result, [...DOM.readerBodyContent.querySelectorAll('.reader-paragraph')]);
}

function wrapSearchFocusRange(paragraph, start, end) {
  const walker = document.createTreeWalker(paragraph, NodeFilter.SHOW_TEXT);
  const segments = [];
  let position = 0;
  let node;
  while ((node = walker.nextNode())) {
    const nodeEnd = position + node.textContent.length;
    const localStart = Math.max(0, start - position);
    const localEnd = Math.min(node.textContent.length, end - position);
    if (localEnd > localStart) segments.push({ node, localStart, localEnd });
    position = nodeEnd;
    if (position >= end) break;
  }
  if (!segments.length) return null;

  const wrappers = [];
  for (const segment of segments.reverse()) {
    let selected = segment.node;
    if (segment.localEnd < selected.textContent.length) selected.splitText(segment.localEnd);
    if (segment.localStart > 0) selected = selected.splitText(segment.localStart);
    const wrapper = document.createElement('span');
    wrapper.className = 'search-focus-match';
    wrapper.tabIndex = -1;
    selected.parentNode.insertBefore(wrapper, selected);
    wrapper.append(selected);
    wrappers.unshift(wrapper);
  }
  activeSearchFocusNodes = wrappers;
  return wrappers[0] || null;
}

async function navigateToGlobalSearchResult(resultIndex) {
  if (searchNavigationPending || bookmarkNavigationPending) return;
  const result = globalSearchResults?.results?.[resultIndex];
  if (!result) return;
  if (scrollRestoreTimeout) {
    clearTimeout(scrollRestoreTimeout);
    scrollRestoreTimeout = null;
  }
  searchNavigationPending = true;
  clearSearchFocusHighlight();
  closeGlobalSearch();

  try {
    const bookId = Number(result.bookId);
    const chapterIndex = Number(result.chapterIndex);
    const book = await getBook(bookId);
    if (!book) {
      showToast('Tác phẩm chứa kết quả này đã bị xóa.', 'warning');
      return;
    }

    const alreadyAtTarget = DOM.readerView.classList.contains('active')
      && Number(state.currentBook?.id) === bookId
      && Number(state.currentChapterIndex) === chapterIndex
      && Number(state.currentChapter?.chapterIndex) === chapterIndex;
    let loaded = true;
    if (!alreadyAtTarget) {
      if (DOM.readerView.classList.contains('active') && Number(state.currentBook?.id) === bookId) {
        loaded = await loadChapter(chapterIndex, 0, 0);
      } else {
        const savedChapterMatches = Number(book.currentChapterIndex) === chapterIndex;
        loaded = await openBook(bookId, chapterIndex, {
          preserveChunk: savedChapterMatches,
          restoreScroll: false
        });
      }
    }
    if (!loaded) {
      void searchIndexService.invalidateBookIndex(bookId).catch(() => {});
      return; // The existing openBook/loadChapter lifecycle already reported the load failure.
    }
    if (!DOM.readerView.classList.contains('active')
      || Number(state.currentBook?.id) !== bookId
      || Number(state.currentChapterIndex) !== chapterIndex
      || Number(state.currentChapter?.chapterIndex) !== chapterIndex) {
      showToast('Không thể mở vị trí tìm kiếm này.', 'warning');
      return;
    }

    const anchor = findSearchAnchorInReader(result);
    if (!anchor) {
      void searchIndexService.invalidateBookIndex(bookId).catch(() => {});
      showToast('Đoạn văn đã thay đổi hoặc không còn tồn tại. Chỉ mục sẽ được cập nhật.', 'info');
      return;
    }

    const focusNode = wrapSearchFocusRange(anchor.paragraph, anchor.start, anchor.end);
    if (!focusNode) {
      showToast('Không thể định vị đoạn khớp trong nội dung hiện tại.', 'info');
      return;
    }
    readerProgressRebasePending = true;
    focusNode.scrollIntoView({ behavior: 'smooth', block: 'center', inline: 'nearest' });
    focusNode.focus({ preventScroll: true });
    resetProgressBaselineAtParagraph(focusNode);
    noteTrustedKeyboardActivity();
  } catch {
    showToast('Không thể mở kết quả tìm kiếm lúc này.', 'warning');
  } finally {
    searchNavigationPending = false;
  }
}

function showMoreGlobalSearchResults() {
  if (!globalSearchResults || DOM.btnGlobalSearchMore.disabled) return;
  if (globalSearchVisibleCount < globalSearchResults.results.length) {
    globalSearchVisibleCount = Math.min(globalSearchVisibleCount + 50, globalSearchResults.results.length);
    renderGlobalSearchResults();
    return;
  }
  if (globalSearchResults.truncated && globalSearchResults.results.length < MAX_SEARCH_LIMIT) {
    runGlobalSearch(globalSearchUiGeneration, DOM.globalSearchInput.value, globalSearchResults.results.length);
  }
}

function handleGlobalSearchKeydown(event) {
  if (event.key === 'Escape') {
    event.preventDefault();
    event.stopPropagation();
    closeGlobalSearch();
    return;
  }
  if (event.isComposing) return;

  const resultRows = [...DOM.globalSearchResultsList.querySelectorAll('[data-search-result="true"]')];
  const focusedIndex = resultRows.indexOf(event.target);
  if ((event.key === 'Enter' || event.key === 'ArrowDown') && event.target === DOM.globalSearchInput && resultRows.length) {
    event.preventDefault();
    resultRows[0].focus();
    return;
  }
  if (focusedIndex >= 0 && (event.key === 'Enter' || event.key === ' ')) {
    event.preventDefault();
    void navigateToGlobalSearchResult(Number(resultRows[focusedIndex].dataset.searchResultIndex));
    return;
  }
  if ((event.key === 'ArrowDown' || event.key === 'ArrowUp') && focusedIndex >= 0) {
    event.preventDefault();
    if (event.key === 'ArrowUp' && focusedIndex === 0) DOM.globalSearchInput.focus();
    else resultRows[Math.max(0, Math.min(resultRows.length - 1, focusedIndex + (event.key === 'ArrowDown' ? 1 : -1)))].focus();
    return;
  }
  if (event.key === 'Tab') {
    const focusable = [...DOM.globalSearchModal.querySelectorAll('button:not([hidden]), input, [data-search-result="true"]')]
      .filter(element => !element.disabled && element.getAttribute('aria-hidden') !== 'true');
    if (!focusable.length) return;
    const first = focusable[0];
    const last = focusable[focusable.length - 1];
    if (event.shiftKey && event.target === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && event.target === last) {
      event.preventDefault();
      first.focus();
    }
  }
}

/**

    } catch (err) {

    }
  }
}

function flushPendingScrollProgress() {
  if (scrollSaveTimeout && state.currentBook) {
    clearTimeout(scrollSaveTimeout);
    scrollSaveTimeout = null;
    recordCurrentEyeReadingProgress({ explicitJump: readerProgressRebasePending });
    readerProgressRebasePending = false;
    const scrollY = window.scrollY;
    const overallProgress = calculateOverallProgress();
    const currentChunk = state.tts ? (state.tts.currentIndex || 0) : (state.currentBook.currentChunkIndex || 0);
    void updateProgress(state.currentBook.id, state.currentChapterIndex, overallProgress, scrollY, currentChunk);
  }
}

/**
 * Opens a book in the reader view.
 */
export async function openBook(bookId, targetChapterIndex = null, options = {}) {
  try {
    const book = await getBook(bookId);
    if (!book) {
      showToast('Không tìm thấy tác phẩm này trong cơ sở dữ liệu.', 'error');
      return false;
    }

    const switchingBooks = Number(state.currentBook?.id) !== Number(book.id);
    if (switchingBooks && state.currentBook) {
      flushPendingScrollProgress();
      if (scrollRestoreTimeout) {
        clearTimeout(scrollRestoreTimeout);
        scrollRestoreTimeout = null;
      }
      state.tts?.stop();
      state.autoScroll?.stop();
      await endReadingHistorySession(Date.now());
    }
    state.currentBook = book;
    state.chaptersMeta = await getBookChaptersMeta(bookId);
    DOM.tocBookTitle.textContent = book.title;

    // Load per-book TTS customization if saved
    if (book.ttsSettings) {
      const bTts = book.ttsSettings;
      if (bTts.voice) {
        state.settings.ttsVoice = bTts.voice;
        if (DOM.ttsVoiceSelect) DOM.ttsVoiceSelect.value = bTts.voice;
        if (bTts.voice.startsWith('fpt:') || bTts.voice.startsWith('openai:')) {
          state.tts?.setEngine('cloud');
          state.tts?.setCloudConfig({ voice: bTts.voice.split(':')[1] });
        } else {
          state.tts?.setEngine('browser');
          state.tts?.setVoice(bTts.voice);
        }
      }
      if (bTts.rate) {
        const rateVal = parseFloat(bTts.rate);
        state.settings.ttsRate = rateVal;
        state.tts?.setRate(rateVal);
        syncTtsRateSelect(rateVal);
      }
      if (bTts.pitch) {
        const pitchVal = parseFloat(bTts.pitch);
        state.settings.ttsPitch = pitchVal;
        state.tts?.setPitch(pitchVal);
      }
      if (bTts.engine) {
        state.settings.ttsEngine = bTts.engine;
        state.tts?.setEngine(bTts.engine);
      }
    }

    // Determine target chapter: specified or resumed
    const chapterIdx = targetChapterIndex !== null ? targetChapterIndex : (book.currentChapterIndex || 0);

    // Switch view
    DOM.libraryView.classList.remove('active');
    DOM.historyView.classList.remove('active');
    DOM.historyView.hidden = true;
    DOM.readerView.classList.add('active');
    window.scrollTo({ top: 0, behavior: 'instant' });

    const useSavedChunk = targetChapterIndex === null || options.preserveChunk === true;
    const initialChunk = useSavedChunk ? (book.currentChunkIndex || 0) : 0;
    const initialScrollY = options.restoreScroll === false
      ? 0
      : (targetChapterIndex === null ? book.scrollY : 0);
    const loaded = await loadChapter(chapterIdx, initialScrollY, initialChunk);
    if (loaded) {
      await startReadingHistorySession({ bookId: book.id, chapterIndex: chapterIdx, timestamp: Date.now() });
      if (!readerProgressRebasePending) recordCurrentEyeReadingProgress({ explicitJump: true });
      lastReadingActivitySignalAt = Date.now();
    }
    return loaded;

  } catch (err) {
    console.error('Lỗi mở sách:', err);
    showToast('Lỗi mở tác phẩm: ' + err.message, 'error');
    return false;
  }
}

/**
 * Loads and displays a specific chapter.
 */
async function loadChapter(chapterIndex, initialScrollY = 0, initialChunkIndex = 0) {
  if (!state.currentBook) return false;

  const totalChapters = state.chaptersMeta.length;
  if (chapterIndex < 0 || chapterIndex >= totalChapters) {
    showToast('Không có chương này.', 'warning');
    return false;
  }

  const currentGeneration = ++chapterLoadGeneration;
  if (scrollRestoreTimeout) {
    clearTimeout(scrollRestoreTimeout);
    scrollRestoreTimeout = null;
  }
  flushPendingScrollProgress();

  const previousChapterIndex = state.currentChapterIndex;
  // Stop active TTS & AutoScroll
  if (state.tts) {
    state.tts.stop();
  }
  if (state.autoScroll) {
    state.autoScroll.stop();
  }
  if (autoScrollNextTimer) {
    clearTimeout(autoScrollNextTimer);
    autoScrollNextTimer = null;
  }
  hideSelectionToolbar();

  state.currentChapterIndex = chapterIndex;
  const chapterData = await getChapter(state.currentBook.id, chapterIndex);
  if (currentGeneration !== chapterLoadGeneration) return false;
  if (!chapterData) {
    showToast('Không thể nạp nội dung chương.', 'error');
    return false;
  }

  state.currentChapter = chapterData;

  // Update header text
  DOM.readerNavTitle.textContent = `${state.currentBook.title} - ${chapterData.title}`;
  DOM.readerChapterTitle.textContent = chapterData.title;

  // Sync MediaSession Notification banner metadata
  if (state.mediaSession) {
    state.mediaSession.updateMetadata({
      title: chapterData.title,
      bookTitle: state.currentBook.title,
      author: state.currentBook.author
    });
  }

  // Two-Layer Text Sanitization Pipeline: Global Cleaner + Custom Dictionary Rules
  const rules = await getCustomRules(state.currentBook.id);
  state.customRules = rules;
  const cleanedContent = cleanChapterText(chapterData.content, {
    rules,
    enableCleaner: state.enableGlobalCleaner
  });

  const words = countWords(cleanedContent);
  const minutes = Math.max(1, Math.round(words / 220));
  DOM.readerChapterMeta.textContent = `Chương ${chapterIndex + 1}/${totalChapters} · Khoảng ${words.toLocaleString('vi-VN')} từ · Ước tính ${minutes} phút đọc`;

  // Update nav buttons disabled state
  DOM.btnPrevChapterNav.disabled = chapterIndex === 0;
  DOM.btnBottomPrevCh.disabled = chapterIndex === 0;
  DOM.btnNextChapterNav.disabled = chapterIndex >= totalChapters - 1;
  DOM.btnBottomNextCh.disabled = chapterIndex >= totalChapters - 1;

  // Render paragraphs and interactive TTS chunks with cleaned content
  renderChapterContent(cleanedContent);
  logicalProgressMap = buildLogicalProgressMap(words);
  activeProgressTtsChunkIndex = null;
  lastCompletedProgressTtsChunkIndex = null;

  // Load cleaned text into TTS engine
  currentTtsChapterIndex = chapterIndex;
  state.tts.loadText(cleanedContent);

  // Apply saved text highlights to reader
  await loadAndApplyChapterHighlights();
  if (currentGeneration !== chapterLoadGeneration) return false;
  await refreshQuotesHubCount();
  if (currentGeneration !== chapterLoadGeneration) return false;

  // Setup Dock Scrubber & Progress Label
  const totalChunks = state.tts.chunks.length;
  const validChunk = Math.min(Math.max(0, initialChunkIndex), Math.max(0, totalChunks - 1));

  if (state.tts) {
    state.tts.jumpToIndex(validChunk, false);
  }

  if (DOM.dockScrubber) {
    DOM.dockScrubber.min = 0;
    DOM.dockScrubber.max = Math.max(0, totalChunks - 1);
    DOM.dockScrubber.value = validChunk;
  }
  if (DOM.dockProgressLabel) {
    const chunkPercent = totalChunks > 0 ? Math.round(((validChunk + 1) / totalChunks) * 100) : 0;
    DOM.dockProgressLabel.textContent = `Chương ${chapterIndex + 1} · Đoạn ${validChunk + 1}/${totalChunks} (${chunkPercent}%)`;
  }
  if (DOM.dockFontSizeLabel) {
    DOM.dockFontSizeLabel.textContent = `${state.settings.fontSize}px`;
  }
  if (DOM.miniFabBadge) {
    const percent = totalChunks > 0 ? Math.round(((validChunk + 1) / totalChunks) * 100) : 0;
    DOM.miniFabBadge.textContent = `${percent}%`;
  }
  if (DOM.btnDockAutoscroll) {
    DOM.btnDockAutoscroll.classList.toggle('active', state.settings.ttsAutoScroll);
  }
  if (DOM.btnDockPrevCh) {
    DOM.btnDockPrevCh.disabled = chapterIndex === 0;
  }
  if (DOM.btnDockNextCh) {
    DOM.btnDockNextCh.disabled = chapterIndex >= totalChapters - 1;
  }

  if (validChunk > 0) {
    highlightSentenceChunk(validChunk);
    if (state.tts.chunks[validChunk] && DOM.ttsPreviewText) {
      DOM.ttsPreviewText.textContent = state.tts.chunks[validChunk];
    }
  }

  // Ensure dock or mini fab is visible when reader is loaded
  if (state.isDockMinimized) {
    if (DOM.ttsDock) DOM.ttsDock.style.display = 'none';
    if (DOM.readerMiniFab) DOM.readerMiniFab.style.display = 'flex';
  } else {
    if (DOM.ttsDock) DOM.ttsDock.style.display = 'flex';
    if (DOM.readerMiniFab) DOM.readerMiniFab.style.display = 'none';
  }

  // Render ToC list
  renderTocDrawerList();
  const bookBookmarks = await refreshBookmarksCount(state.currentBook.id);
  if (DOM.tabViewBookmarks?.style.display !== 'none') {
    await renderBookmarkHubList(bookBookmarks);
  }

  // Scroll to position
  if (initialScrollY > 50) {
    readerProgressRebasePending = true;
    scrollRestoreTimeout = setTimeout(() => {
      if (currentGeneration === chapterLoadGeneration) {
        window.scrollTo({ top: initialScrollY, behavior: 'smooth' });
      }
      scrollRestoreTimeout = null;
    }, 150);
  } else {
    window.scrollTo({ top: 0, behavior: 'instant' });
    readerProgressRebasePending = false;
    recordCurrentEyeReadingProgress({ explicitJump: true });
  }

  // Update progress in DB
  await updateProgress(state.currentBook.id, chapterIndex, calculateOverallProgress(), initialScrollY, validChunk);
  if (currentGeneration !== chapterLoadGeneration) return false;
  if (previousChapterIndex !== null && previousChapterIndex !== undefined && Number.isSafeInteger(Number(previousChapterIndex)) && previousChapterIndex !== chapterIndex) {
    markReadingHistoryChapterLeave(previousChapterIndex, Date.now());
    markReadingHistoryChapterEnter(chapterIndex, Date.now());
  }
  updateProgressBar();
  return true;
}

function buildLogicalProgressMap(totalWords) {
  const paragraphs = new Map();
  const chunks = new Map();
  let cursor = 0;
  const renderedParagraphs = DOM.readerBodyContent.querySelectorAll('.reader-paragraph[data-para-index]');
  for (const paragraph of renderedParagraphs) {
    const paragraphIndex = Number(paragraph.getAttribute('data-para-index'));
    if (!Number.isSafeInteger(paragraphIndex) || paragraphIndex < 0) continue;
    const paragraphWordCount = countWords(paragraph.textContent || '');
    const startWord = cursor;
    let chunkCursor = startWord;
    for (const element of paragraph.querySelectorAll('.tts-chunk[data-chunk-index]')) {
      const chunkIndex = Number(element.getAttribute('data-chunk-index'));
      const chunkWords = countWords(element.textContent || '');
      if (!Number.isSafeInteger(chunkIndex) || chunkIndex < 0 || chunkWords <= 0) continue;
      chunks.set(chunkIndex, { startWord: chunkCursor, endWord: chunkCursor + chunkWords });
      chunkCursor += chunkWords;
    }
    paragraphs.set(paragraphIndex, {
      startWord,
      endWord: startWord + paragraphWordCount,
      midpointWord: startWord + Math.floor(paragraphWordCount / 2)
    });
    cursor += paragraphWordCount;
  }

  const estimatedWords = Math.max(1, Math.floor(Number(totalWords) || cursor));
  const scale = cursor > 0 ? estimatedWords / cursor : 1;
  const scaleOffset = offset => Math.min(estimatedWords, Math.floor(offset * scale));
  for (const [index, range] of paragraphs) {
    paragraphs.set(index, {
      startWord: scaleOffset(range.startWord),
      endWord: scaleOffset(range.endWord),
      midpointWord: scaleOffset(range.midpointWord)
    });
  }
  for (const [index, range] of chunks) {
    chunks.set(index, {
      startWord: scaleOffset(range.startWord),
      endWord: scaleOffset(range.endWord)
    });
  }
  return { chapterIndex: state.currentChapterIndex, totalWords: estimatedWords, paragraphs, chunks };
}

function recordCurrentEyeReadingProgress({ explicitJump = false } = {}) {
  const progress = logicalProgressMap;
  if (!progress || actualTtsPlaybackActive || document.visibilityState === 'hidden') return false;
  const anchorY = Math.max(1, Math.min(window.innerHeight - 1, Math.round(window.innerHeight * 0.4)));
  const anchorXs = [0.5, 0.35, 0.65].map(ratio => Math.round(window.innerWidth * ratio));
  let paragraph = null;
  for (const x of anchorXs) {
    const target = document.elementFromPoint(x, anchorY);
    const candidate = target?.closest?.('.reader-paragraph[data-para-index]');
    if (candidate && DOM.readerBodyContent.contains(candidate)) {
      paragraph = candidate;
      break;
    }
  }
  if (!paragraph) return false;
  const paragraphIndex = Number(paragraph.getAttribute('data-para-index'));
  const logical = progress.paragraphs.get(paragraphIndex);
  if (!logical) return false;
  return recordReadingHistoryChapterProgress({
    chapterIndex: progress.chapterIndex,
    position: logical.midpointWord,
    totalWords: progress.totalWords,
    explicitJump,
    timestamp: Date.now()
  });
}

function resetProgressBaselineAtParagraph(paragraph) {
  lastCompletedProgressTtsChunkIndex = null;
  activeProgressTtsChunkIndex = null;
  const progress = logicalProgressMap;
  const paragraphIndex = Number(paragraph?.closest?.('.reader-paragraph[data-para-index]')
    ?.getAttribute?.('data-para-index'));
  const logical = progress?.paragraphs.get(paragraphIndex);
  if (!progress || !logical) return false;
  return recordReadingHistoryChapterProgress({
    chapterIndex: progress.chapterIndex,
    position: logical.midpointWord,
    totalWords: progress.totalWords,
    explicitJump: true,
    timestamp: Date.now()
  });
}

/**
 * Renders chapter paragraphs with span-wrapped TTS sentence chunks.
 */
function renderChapterContent(rawContent) {
  if (!rawContent || !rawContent.trim()) {
    DOM.readerBodyContent.innerHTML = '<p class="reader-paragraph">Nội dung chương trống.</p>';
    return;
  }

  const paragraphs = rawContent.split(/\n+/);
  let chunkGlobalIndex = 0;
  let html = '';

  for (let pIdx = 0; pIdx < paragraphs.length; pIdx++) {
    const para = paragraphs[pIdx];
    const trimmed = para.trim();
    if (!trimmed) continue;

    // Sentence splitting matching TTS logic
    const sentenceRegex = /[^.!?…;:\n]+[.!?…;:]*|\S+/g;
    const sentences = trimmed.match(sentenceRegex) || [trimmed];

    let paraInnerHtml = '';
    for (let s of sentences) {
      const sentenceText = s.trim();
      if (!sentenceText) continue;

      if (sentenceText.length > 220) {
        const subClauses = sentenceText.split(/,\s+/);
        let tempBuf = '';
        for (const sub of subClauses) {
          if ((tempBuf + sub).length < 180) {
            tempBuf += (tempBuf ? ', ' : '') + sub;
          } else {
            if (tempBuf) {
              paraInnerHtml += `<span class="tts-chunk" data-chunk-index="${chunkGlobalIndex++}" title="Nhấp để nghe từ câu này">${escapeHTML(tempBuf.trim())} </span>`;
            }
            tempBuf = sub;
          }
        }
        if (tempBuf) {
          paraInnerHtml += `<span class="tts-chunk" data-chunk-index="${chunkGlobalIndex++}" title="Nhấp để nghe từ câu này">${escapeHTML(tempBuf.trim())} </span>`;
        }
      } else {
        paraInnerHtml += `<span class="tts-chunk" data-chunk-index="${chunkGlobalIndex++}" title="Nhấp để nghe từ câu này">${escapeHTML(sentenceText)} </span>`;
      }
    }

    html += `<p class="reader-paragraph" data-para-index="${pIdx}">${paraInnerHtml}</p>`;
  }

  DOM.readerBodyContent.innerHTML = html;

  // Add click-to-speak listeners on each sentence
  DOM.readerBodyContent.querySelectorAll('.tts-chunk').forEach(span => {
    span.addEventListener('click', () => {
      const idx = parseInt(span.getAttribute('data-chunk-index'), 10);
      if (!isNaN(idx)) {
        DOM.ttsDock.style.display = 'flex';
        state.tts.jumpToIndex(idx, true);
      }
    });
  });
}

/**
 * Navigates to prev/next chapter relative to current index.
 */
async function navigateChapter(delta, { activity = true } = {}) {
  if (!state.currentBook) return;
  if (isChapterSwitching) {
    console.warn('>>> [VietReader] navigateChapter: Đang chuyển chương, bỏ qua yêu cầu chuyển chương đồng thời.');
    return;
  }
  const newIndex = state.currentChapterIndex + delta;
  if (newIndex >= 0 && newIndex < state.chaptersMeta.length) {
    recordCurrentEyeReadingProgress({ explicitJump: readerProgressRebasePending });
    readerProgressRebasePending = false;
    if (delta > 0 && isReadingHistoryChapterNearEnd(state.currentChapterIndex)) {
      markReadingHistoryChapterCompleted(state.currentChapterIndex, Date.now());
    }
    if (activity) noteTrustedKeyboardActivity();
    await loadChapter(newIndex, 0);
  }
}

/**
 * Handles window scroll to update reading progress bars and persist scroll position.
 */
function handleWindowScroll() {
  if (!DOM.readerView.classList.contains('active')) return;

  updateProgressBar();

  const targetBookId = state.currentBook?.id;
  const targetChapterIndex = state.currentChapterIndex;

  // Debounced auto-save scrollY to IndexedDB
  clearTimeout(scrollSaveTimeout);
  scrollSaveTimeout = setTimeout(() => {
    if (state.currentBook && Number(state.currentBook.id) === Number(targetBookId) && Number(state.currentChapterIndex) === Number(targetChapterIndex)) {
      recordCurrentEyeReadingProgress({ explicitJump: readerProgressRebasePending });
      readerProgressRebasePending = false;
      const scrollY = window.scrollY;
      const overallProgress = calculateOverallProgress();
      const currentChunk = state.tts ? (state.tts.currentIndex || 0) : (state.currentBook.currentChunkIndex || 0);
      updateProgress(state.currentBook.id, state.currentChapterIndex, overallProgress, scrollY, currentChunk);
    }
  }, 400);
}

/**
 * Calculates both chapter scroll percentage and overall book completion percentage.
 */
function calculateOverallProgress() {
  if (!state.currentBook || state.chaptersMeta.length === 0) return 0;

  const docHeight = document.documentElement.scrollHeight - window.innerHeight;
  const chapterScrollProgress = docHeight > 0 ? (window.scrollY / docHeight) : 0;
  const clampedChapterProgress = Math.max(0, Math.min(1, chapterScrollProgress));

  const totalChapters = state.chaptersMeta.length;
  // Progress = (currentChapterIndex + chapterScrollFraction) / totalChapters * 100
  const overall = ((state.currentChapterIndex + clampedChapterProgress) / totalChapters) * 100;
  return Math.round(overall);
}

/**
 * Updates top progress bar and percentage pill.
 */
function updateProgressBar() {
  const percent = calculateOverallProgress();
  DOM.topProgressInner.style.width = `${percent}%`;
  DOM.readingPercentageBadge.textContent = `${percent}%`;
}

function analyticsFormatCount(value) {
  const count = Number(value);
  return Number.isFinite(count) && count > 0 ? Math.floor(count).toLocaleString('vi-VN') : '0';
}

function analyticsFormatWords(value) {
  return `≈ ${analyticsFormatCount(value)} từ`;
}

function analyticsFormatDate(dateKey, options = { day: '2-digit', month: '2-digit', year: 'numeric' }) {
  if (typeof dateKey !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(dateKey)) return '';
  const [year, month, day] = dateKey.split('-').map(Number);
  return new Date(year, month - 1, day).toLocaleDateString('vi-VN', options);
}

}

function renderAnalyticsBars(container, rows, label) {
  if (!container) return;
  const values = Array.isArray(rows) ? rows : [];
  const maximum = Math.max(0, ...values.map(row => Number(row.duration) || 0));
  const bars = values.map((row, index) => {
    const duration = Math.max(0, Number(row.duration) || 0);
    const height = maximum > 0 ? Math.max(3, Math.round(duration / maximum * 100)) : 0;
    const dateLabel = analyticsFormatDate(row.date, { weekday: 'long', day: 'numeric', month: 'long' });
    const accessible = `${dateLabel}: ${formatHistoryDuration(duration)}`;
    const shortDate = values.length <= 7
      ? analyticsFormatDate(row.date, { weekday: 'short' })
      : (index % 5 === 0 ? analyticsFormatDate(row.date, { day: '2-digit' }) : '');
    return `<div class="analytics-bar-column" title="${escapeHTML(accessible)}"><span class="analytics-bar-value" style="height:${height}%" aria-hidden="true"></span><span class="analytics-bar-label" aria-hidden="true">${escapeHTML(shortDate)}</span></div>`;
  }).join('');
  const rangeTotal = values.reduce((sum, row) => sum + (Number(row.duration) || 0), 0);
  container.innerHTML = `<div class="analytics-bars" role="img" aria-label="${escapeHTML(label)}. Tổng ${escapeHTML(formatHistoryDuration(rangeTotal))}.">${bars}</div><p class="analytics-chart-summary">${escapeHTML(label)} · Tổng ${escapeHTML(formatHistoryDuration(rangeTotal))}</p>`;
}

function renderAnalyticsHeatmap(heatmap) {
  const weeks = Array.isArray(heatmap?.weeks) ? heatmap.weeks : [];
  const monthMarks = Array.isArray(heatmap?.monthMarks) ? heatmap.monthMarks : [];
  const count = Math.max(weeks.length, 1);
  const monthHeader = Array.from({ length: count }, (_, index) => {
    const mark = monthMarks.find(item => item.weekIndex === index);
    return `<span>${mark ? escapeHTML(mark.label) : ''}</span>`;
  }).join('');
  const weekdays = ['T2', 'T3', 'T4', 'T5', 'T6', 'T7', 'CN'];
  const rows = weekdays.map((weekday, dayIndex) => {
    const cells = weeks.map(week => {
      const cell = week[dayIndex];
      if (!cell) return '<span class="activity-cell activity-cell-empty" aria-hidden="true"></span>';
      const date = analyticsFormatDate(cell.date);
      const label = `${date}. ${formatHistoryDuration(cell.activeDurationMs)}. ${analyticsFormatWords(cell.wordsReadEstimate)}. ${formatHistoryDuration(cell.ttsDurationMs)} TTS. ${analyticsFormatCount(cell.chaptersCompleted)} chương. ${analyticsFormatCount(cell.booksActive)} truyện.`;
      return `<span class="activity-cell level-${cell.intensity}" title="${escapeHTML(label)}" aria-label="${escapeHTML(label)}" role="img"></span>`;
    }).join('');
    return `<div class="analytics-heatmap-row" style="--heatmap-weeks:${count}"><span class="analytics-weekday" aria-hidden="true">${weekday}</span>${cells}</div>`;
  }).join('');
  DOM.analyticsHeatmap.innerHTML = `<div class="analytics-month-row" style="--heatmap-weeks:${count}"><span></span>${monthHeader}</div>${rows}`;
}

function renderAnalyticsComparison(comparisons) {
  const renderPeriod = (comparison, period, label) => {
    const metric = comparison?.metrics?.activeDurationMs;
    if (!metric) return `<section><strong>${label}</strong><p>Chưa đủ dữ liệu để so sánh hai giai đoạn.</p></section>`;
    const previousLabel = period === '7days' ? '7 ngày trước' : '30 ngày trước';
    let direction = '';
    if (metric.baseline === 'zero') {
      direction = metric.current > 0
        ? 'Bắt đầu có hoạt động trong giai đoạn này.'
        : 'Chưa có hoạt động trong hai giai đoạn.';
    } else if (metric.direction === 'equal') direction = 'Hoạt động tương đương giai đoạn trước.';
    else direction = `${metric.direction === 'increase' ? 'Nhiều hơn' : 'Ít hơn'} ${Math.round(Math.abs(metric.percentChange))}% so với giai đoạn trước.`;
    return `<section><strong>${label}</strong><p>${escapeHTML(period === '7days' ? '7 ngày này' : '30 ngày này')}: ${escapeHTML(formatHistoryDuration(metric.current))}</p><p>${escapeHTML(previousLabel)}: ${escapeHTML(formatHistoryDuration(metric.previous))}</p><p class="analytics-comparison-note">${escapeHTML(direction)}</p></section>`;
  };
  DOM.analyticsComparison.innerHTML = `<div class="analytics-comparisons">${renderPeriod(comparisons?.sevenDay, '7days', 'So sánh 7 ngày')}${renderPeriod(comparisons?.thirtyDay, '30days', 'So sánh 30 ngày')}</div>`;
}

function renderAnalytics(data, period) {
  const totals = data.totals || {};
  const life = data.lifetime?.totals || {};
  const insight = data.insights || {};
  const hasHistory = [life.activeDurationMs, life.ttsDurationMs, life.wordsReadEstimate, life.chaptersCompleted]
    .some(value => Number(value) > 0);
  DOM.analyticsEmpty.hidden = hasHistory;

function formatHistoryDate(dateKey) {
  if (typeof dateKey !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(dateKey)) return '';
  const [, month, day] = dateKey.split('-');
  return `${day}/${month}/${dateKey.slice(0, 4)}`;
}

function formatHistoryDuration(value) {
  const milliseconds = Number(value);
  if (!Number.isFinite(milliseconds) || milliseconds <= 0) return '0 phút';
  const minutes = Math.floor(milliseconds / 60_000);
  if (minutes < 1) return 'Dưới 1 phút';
  const hours = Math.floor(minutes / 60);
  const rest = minutes % 60;
  if (!hours) return `${minutes} phút`;
  return rest ? `${hours} giờ ${rest} phút` : `${hours} giờ`;
}

function formatHistoryCount(value) {
  const number = Number(value);
  return Number.isFinite(number) && number > 0
    ? Math.floor(number).toLocaleString('vi-VN')
    : '0';
}

function historyLocalDay(timestamp) {
  const value = Number(timestamp);
  return Number.isFinite(value) && Math.abs(value) <= 8.64e15
    ? getLocalCalendarRange('today', value)?.to ?? ''
    : '';
}

function formatHistorySessionTime(session) {
  const start = Number(session.startedAt);
  if (!Number.isFinite(start) || Math.abs(start) > 8.64e15) return 'Không rõ thời gian';
  const startLabel = new Date(start).toLocaleTimeString('vi-VN', { hour: '2-digit', minute: '2-digit' });
  if (session.endedAt === null || session.endedAt === undefined) return `${startLabel} → Đang đọc`;
  const end = Number(session.endedAt);
  if (!Number.isFinite(end) || Math.abs(end) > 8.64e15) return startLabel;
  const endLabel = new Date(end).toLocaleTimeString('vi-VN', { hour: '2-digit', minute: '2-digit' });
  const nextDay = historyLocalDay(start) !== historyLocalDay(end);
  return `${startLabel} → ${endLabel}${nextDay ? ' (+1 ngày)' : ''}`;
}

function formatHistoryChapterRange(session, chapters) {
  const chapterTitle = index => {
    const chapter = chapters.find(item => Number(item.chapterIndex) === Number(index));
    return chapter?.title?.trim() || `Chương ${Number(index) + 1}`;
  };
  if (session.startChapterIndex === null || session.startChapterIndex === undefined
    || !Number.isSafeInteger(Number(session.startChapterIndex))) return 'Chưa rõ chương';
  const start = chapterTitle(session.startChapterIndex);
  const endIndex = session.endChapterIndex !== null && session.endChapterIndex !== undefined
    && Number.isSafeInteger(Number(session.endChapterIndex))
    ? Number(session.endChapterIndex)
    : Number(session.startChapterIndex);
  const end = chapterTitle(endIndex);
  return start === end ? start : `${start} → ${end}`;
}

async function loadHistoryView() {
  const generation = ++historyLoadGeneration;
  const todayKey = getLocalCalendarRange('today').to;
  if (todayKey !== historyStatsDateKey) {
    historyStatsDateKey = todayKey;
    historyStatsCache.clear();
  }
  const range = getLocalCalendarRange(historyPeriod);
  DOM.historyLoading.hidden = false;
  DOM.historyStatus.textContent = 'Đang tải lịch sử đọc';
  DOM.historyEmpty.hidden = true;
  DOM.historyDashboard.hidden = true;
  DOM.historyTimeline.hidden = true;
  try {
    let stats = historyStatsCache.get(historyPeriod);
    if (!stats) {
      // Finish the full-store aggregation before fetching the timeline page,
      // so both session arrays are not resident at the same time.
      stats = await getReadingHistoryRangeStats({ period: historyPeriod });
      historyStatsCache.set(historyPeriod, stats);
    }
    const [groups, allHistory, books] = await Promise.all([
      getReadingHistoryTimeline({ from: range.from, to: range.to, limit: historyVisibleLimit }),
      getReadingHistory({ limit: 1 }),
      getAllBooks()
    ]);
    if (generation !== historyLoadGeneration || !DOM.historyView.classList.contains('active')) return;
    DOM.historyLoading.hidden = true;
    DOM.btnClearHistory.hidden = allHistory.length === 0;
    const sessionCount = groups.reduce((count, group) => count + group.sessions.length, 0);
    if (!sessionCount) {
      const hasOlderHistory = allHistory.length > 0;
      DOM.historyEmpty.querySelector('h2').textContent = hasOlderHistory
        ? 'Chưa có phiên đọc trong khoảng này'
        : 'Chưa có lịch sử đọc';

      DOM.historyEmpty.hidden = false;
      DOM.historyStatus.textContent = 'Không có phiên đọc trong khoảng thời gian đã chọn';
      return;
    }

    const bookMap = new Map(books.map(book => [Number(book.id), book]));
    const sessions = groups.flatMap(group => group.sessions);
    const bookIds = [...new Set(sessions.map(session => Number(session.bookId)).filter(Number.isSafeInteger))];
    const chapterResults = await Promise.all(bookIds.map(bookId => {
      if (!historyChapterCache.has(bookId)) {
        historyChapterCache.set(bookId, getBookChaptersMeta(bookId).catch(() => []));
      }
      return historyChapterCache.get(bookId);
    }));
    chapterResults.forEach((chapters, index) => historyChapterCache.set(bookIds[index], chapters));
    if (generation !== historyLoadGeneration || !DOM.historyView.classList.contains('active')) return;

    const labels = {
      today: `Hôm nay · ${formatHistoryDate(range.to)}`,
      '7days': `${formatHistoryDate(range.from)} – ${formatHistoryDate(range.to)}`,
      '30days': `${formatHistoryDate(range.from)} – ${formatHistoryDate(range.to)}`

    DOM.historyMetrics.innerHTML = [
      ['Thời gian đọc / nghe chủ động', formatHistoryDuration(stats.activeDurationMs)],
      ['Số từ ước tính', `≈ ${formatHistoryCount(stats.wordsReadEstimate)} từ`],
      ['Thời gian nghe TTS', formatHistoryDuration(stats.ttsDurationMs)],
      ['Chương đã hoàn thành', formatHistoryCount(stats.chaptersCompleted)],
      ['Sách có hoạt động', formatHistoryCount(stats.distinctBooks)]
    ].map(([label, value]) => `<div class="history-metric"><dt>${escapeHTML(label)}</dt><dd>${escapeHTML(value)}</dd></div>`).join('');
    DOM.historyGroups.innerHTML = groups.map(group => {
      const cards = group.sessions.map(session => {
        const bookId = Number(session.bookId);
        const book = bookMap.get(bookId);
        const title = book?.title || 'Sách đã lưu';
        const chapters = historyChapterCache.get(bookId) || [];
        const contribution = session.dayContribution || {};
        const tts = Number(contribution.ttsDurationMs) || 0;

        const completed = Number(contribution.chaptersCompleted) || 0;

            <span>Đọc / nghe: ${escapeHTML(formatHistoryDuration(contribution.activeDurationMs))}</span>

    showToast('Không thể tải lịch sử đọc lúc này.', 'error');

  }

async function handleHistoryTimelineClick(event) {
  const deleteButton = event.target.closest('.history-session-delete');
  if (deleteButton) {
    const sessionId = deleteButton.closest('.history-session')?.dataset.sessionId;
    if (!sessionId) return;
    const deleted = await deleteReadingHistorySession(sessionId);
    if (!deleted) {
      showToast('Không thể xóa phiên đọc lúc này.', 'error');
      return;
    }
    historyStatsCache.clear();
    showToast('Đã xóa phiên đọc khỏi lịch sử.', 'info');
    await loadHistoryView();
    (DOM.historyTimeline.hidden ? DOM.historyTitle : DOM.historyTimelineTitle).focus();
    return;
  }
  const openButton = event.target.closest('.history-session-open');
  if (openButton) await openBook(Number(openButton.dataset.bookId));
}

    const dump = await exportDatabase();

    a.download = `vietreader_backup_${new Date().toISOString().slice(0, 10)}.json`;

  } catch (err) {
    console.error('Lỗi xuất dữ liệu:', err);
    showToast('Lỗi khi xuất dữ liệu: ' + err.message, 'error');
  }

  const file = e.target.files[0];

    const data = JSON.parse(text);

    const clearExisting = await showCustomConfirm(

  }

}

/**
 * Custom Promise-based modal confirmation (Complies with Zero Native Confirm standard).
 */
function showCustomConfirm(title, message) {
  return new Promise((resolve) => {
    const returnFocus = document.activeElement;
    DOM.confirmModalTitle.textContent = title;
    DOM.confirmModalMessage.textContent = message;
    DOM.confirmModal.classList.add('open');

    const handleOk = () => {
      cleanup();
      resolve(true);
    };

    const handleCancel = () => {
      cleanup();
      resolve(false);
    };
    const handleKeydown = event => {
      if (event.key === 'Escape') {
        event.preventDefault();
        handleCancel();
      } else if (event.key === 'Tab' && event.shiftKey && document.activeElement === DOM.confirmModalBtnCancel) {
        event.preventDefault();
        DOM.confirmModalBtnOk.focus();
      } else if (event.key === 'Tab' && !event.shiftKey && document.activeElement === DOM.confirmModalBtnOk) {
        event.preventDefault();
        DOM.confirmModalBtnCancel.focus();
      }
    };

    const cleanup = () => {
      DOM.confirmModalBtnOk.removeEventListener('click', handleOk);
      DOM.confirmModalBtnCancel.removeEventListener('click', handleCancel);
      DOM.confirmModal.removeEventListener('keydown', handleKeydown);
      DOM.confirmModal.classList.remove('open');
      if (returnFocus?.isConnected) returnFocus.focus();
    };

    DOM.confirmModalBtnOk.addEventListener('click', handleOk);
    DOM.confirmModalBtnCancel.addEventListener('click', handleCancel);
    DOM.confirmModal.addEventListener('keydown', handleKeydown);
    DOM.confirmModalBtnCancel.focus();
  });
}

/**
 * Formats a timestamp into friendly relative time in Vietnamese.
 */
function formatTimeAgo(timestamp) {
  if (!timestamp) return 'Chưa đọc';
  const elapsed = Math.floor((Date.now() - timestamp) / 1000);

  if (elapsed < 60) return 'Vừa xong';
  if (elapsed < 3600) return `${Math.floor(elapsed / 60)} phút trước`;
  if (elapsed < 86400) return `${Math.floor(elapsed / 3600)} giờ trước`;
  if (elapsed < 604800) return `${Math.floor(elapsed / 86400)} ngày trước`;
  return new Date(timestamp).toLocaleDateString('vi-VN');
}

/**
 * Escapes HTML characters to prevent XSS.
 */
function escapeHTML(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

/**
 * Opens TTS Multi-Engine & Voice Tuning Settings modal.
 */
function openTtsSettingsModal() {
  const isCloud = state.settings.ttsEngine === 'cloud';
  switchEngineTab(isCloud ? 'cloud' : 'browser');

  DOM.sliderTtsPitch.value = state.settings.ttsPitch || 1.0;
  updatePitchLabel(state.settings.ttsPitch || 1.0);

  DOM.sliderModalTtsRate.value = state.settings.ttsRate || 1.0;
  DOM.labelModalTtsRate.textContent = `${(state.settings.ttsRate || 1.0).toFixed(1)}x`;

  DOM.cloudVoiceSelect.value = state.settings.ttsCloudVoice || 'banmai';

  state.settings.ttsEngine = engine;
  state.tts?.setEngine(engine);

  DOM.tabEngineBrowser.classList.toggle('active', !isCloud);

function updatePitchLabel(val) {
  const p = parseFloat(val);
  let desc = 'Chuẩn';
  if (p <= 0.65) desc = 'Rất trầm / Già dặn';
  else if (p < 0.95) desc = 'Trầm ấm';
  else if (p > 1.25) desc = 'Thanh cao';
  else if (p > 1.05) desc = 'Trẻ trung';
  DOM.labelTtsPitch.textContent = `${p.toFixed(2)} (${desc})`;
}

/**
 * Initializes Hands-Free Auto-Scroll Engine.
 */
function initAutoScrollEngine() {
  state.autoScroll = new AutoScrollEngine({
    speed: 40,
    resumeDelay: 2000,
    onStateChange: ({ isActive, isPaused, speed }) => {
      if (DOM.btnHandsfreeScroll) {
        DOM.btnHandsfreeScroll.classList.toggle('active', isActive);
        if (isActive) {
          DOM.labelHandsfreeBtn.textContent = isPaused ? 'Tạm dừng (2s)...' : 'Đang cuộn (A)';
        } else {
          DOM.labelHandsfreeBtn.textContent = 'Cuộn rảnh tay';
        }
      }
      if (DOM.autoscrollSpeedControls) {
        DOM.autoscrollSpeedControls.style.display = isActive ? 'flex' : 'none';
      }
      if (DOM.labelScrollSpeed) {
        DOM.labelScrollSpeed.textContent = `${speed}px/s`;
      }
      if (!lastHistoryAutoScrollState || lastHistoryAutoScrollState.isActive !== isActive
        || lastHistoryAutoScrollState.isPaused !== isPaused) {
        lastHistoryAutoScrollState = { isActive, isPaused };
        void setReadingHistoryAutoScrollState({ isActive, isPaused, timestamp: Date.now() })
          .catch(error => console.warn('Reading History AutoScroll signal failed:', error));
      }
    },
    onReachBottom: () => {
      // Nếu TTS đang phát hoặc đang trong tiến trình chuyển chương, nhường toàn quyền điều phối cho TTS
      // để chống trùng lặp sự kiện chuyển chương (duplicate chapter transition) và lặp lịch sử đọc.
      if (state.tts?.isPlaying || isChapterSwitching) {
        state.autoScroll?.stop();
        return;
      }

      if (document.visibilityState !== 'hidden') {
        markReadingHistoryChapterCompleted(state.currentChapterIndex, Date.now());
      }
      if (state.sleepTimer?.isStopAtChapterEnd && state.sleepTimer.isStopAtChapterEnd()) {
        state.sleepTimer?.notifyChapterEnd?.();
        state.sleepTimer?.consumeChapterEndTrigger?.();
        showToast('Đã cuộn đến hết chương. Hẹn giờ ngủ đã dừng tự động chuyển chương.', 'info');
        return;
      }
      if (state.currentChapterIndex < state.chaptersMeta.length - 1) {
        showToast('Đã cuộn đến hết chương. Tự động chuyển chương sau 1.5s...', 'info');
        if (autoScrollNextTimer) {
          clearTimeout(autoScrollNextTimer);
        }
        autoScrollNextTimer = setTimeout(() => {
          autoScrollNextTimer = null;
          if (!isChapterSwitching && (!state.tts || !state.tts.isPlaying)) {
            navigateChapter(1, { activity: false });
          }
        }, 1500);
      } else {
        showToast('Đã đọc đến cuối tác phẩm!', 'info');
      }
    }
  });

  state.autoScroll.bindListeners(DOM.ttsDock);
}

/**
 * Toggles Hands-Free Auto-Scroll on or off.
 */
export function toggleHandsFreeAutoScroll() {
  if (!DOM.readerView.classList.contains('active')) return;
  if (!state.autoScroll) return;

  const isActive = state.autoScroll.toggle();
  if (isActive) {
    showToast(`Bật cuộn rảnh tay (${state.autoScroll.speed}px/s) - Phím A để dừng`, 'success');
  } else {
    showToast('Đã tắt cuộn rảnh tay', 'info');
  }
}

/**
 * Handles mouseup and touchend in reader content to detect text selections.
 */
function handleTextSelectionEnd(e) {
  // If clicking inside toolbar or popover, don't trigger selection change
  if (e.target.closest('#selection-toolbar') || e.target.closest('#hl-note-popover')) {
    return;
  }

  // Small delay to allow browser selection range to settle
  setTimeout(() => {
    const sel = window.getSelection();
    if (!sel || sel.isCollapsed) {
      if (!state.activeHighlightId) {
        hideSelectionToolbar();
      }
      return;
    }

    const text = sel.toString().trim();
    if (text.length < 2) {
      if (!state.activeHighlightId) {
        hideSelectionToolbar();
      }
      return;
    }

    // Verify selection is within reader body content
    const range = sel.getRangeAt(0);
    const container = range.commonAncestorContainer;
    if (!DOM.readerBodyContent.contains(container.nodeType === 1 ? container : container.parentNode)) {
      hideSelectionToolbar();
      return;
    }

    state.activeSelectionRange = range.cloneRange();
    state.activeSelectedText = text;
    state.activeHighlightId = null;

    // Show toolbar without delete button
    if (DOM.btnHlDelete) DOM.btnHlDelete.style.display = 'none';
    positionFloatingToolbar(range.getBoundingClientRect());
  }, 30);
}

/**
 * Listens to selection change globally to dismiss toolbar when cleared.
 */
function handleSelectionChange() {
  const sel = window.getSelection();
  if ((!sel || sel.isCollapsed) && !state.activeHighlightId && DOM.hlNotePopover.style.display === 'none') {
    hideSelectionToolbar();
  }
}

/**
 * Positions floating toolbar right above the target bounding rectangle.
 * @param {DOMRect} rect
 */
function positionFloatingToolbar(rect) {
  if (!DOM.selectionToolbar || !rect) return;

  DOM.selectionToolbar.style.display = 'flex';
  const toolbarWidth = DOM.selectionToolbar.offsetWidth || 280;
  const toolbarHeight = DOM.selectionToolbar.offsetHeight || 38;

  let top = window.scrollY + rect.top - toolbarHeight - 10;
  let left = window.scrollX + rect.left + (rect.width / 2) - (toolbarWidth / 2);

  // Viewport bounds checking
  if (top < window.scrollY + 10) {
    top = window.scrollY + rect.bottom + 10; // Flip below if too close to top
  }
  const minLeft = window.scrollX + 10;
  const maxLeft = window.scrollX + window.innerWidth - toolbarWidth - 10;
  left = Math.max(minLeft, Math.min(maxLeft, left));

  DOM.selectionToolbar.style.top = `${top}px`;
  DOM.selectionToolbar.style.left = `${left}px`;
}

/**
 * Handles clicking directly on an existing highlight tag in reader.
 */
async function handleReaderBodyClick(e) {
  const mark = e.target.closest('.reader-highlight');
  if (!mark) return;

  e.stopPropagation();
  const hlId = mark.getAttribute('data-highlight-id');
  if (!hlId) return;

  state.activeHighlightId = Number(hlId);
  const found = state.currentHighlights.find(h => Number(h.id) === Number(hlId));
  if (found) {
    state.activeSelectedText = found.selectedText;
  } else {
    state.activeSelectedText = mark.textContent.trim();

  positionFloatingToolbar(mark.getBoundingClientRect());
}

/**
 * Opens inline popover for typing note/thoughts.
 */

  if (!DOM.hlNotePopover || !DOM.selectionToolbar) return;

  const toolbarRect = DOM.selectionToolbar.getBoundingClientRect();
  DOM.hlNotePopover.style.display = 'flex';

  let top = window.scrollY + toolbarRect.bottom + 8;
  let left = window.scrollX + toolbarRect.left;

  // Viewport bounds clamp
  const maxLeft = window.scrollX + window.innerWidth - 305;
  left = Math.max(window.scrollX + 10, Math.min(maxLeft, left));

  DOM.hlNotePopover.style.top = `${top}px`;
  DOM.hlNotePopover.style.left = `${left}px`;

  // Pre-fill existing note if editing
  if (state.activeHighlightId) {
    const found = state.currentHighlights.find(h => Number(h.id) === Number(state.activeHighlightId));

    DOM.hlNoteTextarea.value = '';
    DOM.notePopoverTitle.textContent = 'Thêm ghi chú cho trích dẫn';
  }

  DOM.hlNoteTextarea.focus();
}

async function handleCreateBookmark() {
  const book = state.currentBook;
  if (!book) return;

  const position = captureCurrentReadingPosition({
    bookId: book.id,
    chapterIndex: state.currentChapterIndex,
    readerBodyContent: DOM.readerBodyContent,
    tts: state.tts,
    readerSubHeader: document.querySelector('.reader-sub-header'),
    ttsDock: DOM.ttsDock,
    windowRef: window
  });
  if (!position) {
    showToast('Không thể xác định vị trí đọc hiện tại.', 'warning');
    return;
  }

  const existing = (await getBookBookmarks(book.id)).some(item =>
    Number(item.chapterIndex) === position.chapterIndex
    && Number(item.paragraphIndex) === position.paragraphIndex
  );
  const saved = await createBookmark(position);
  if (!saved) {
    showToast('Không thể lưu bookmark lúc này.', 'error');
    return;
  }

  if (existing) showToast('Vị trí đọc này đã có bookmark.', 'info');
  else showToast('✓ Đã đánh dấu vị trí đọc', 'success');
  await refreshBookmarksCount(book.id);
  if (DOM.tabViewBookmarks?.style.display !== 'none') await renderBookmarkHubList();
}

async function refreshBookmarksCount(bookId = state.currentBook?.id) {
  if (!DOM.bookmarksCountBadge || !bookId) {
    if (DOM.bookmarksCountBadge) DOM.bookmarksCountBadge.textContent = '0';
    return [];
  }
  const bookmarks = await getBookBookmarks(bookId);
  if (Number(state.currentBook?.id) === Number(bookId)) {
    DOM.bookmarksCountBadge.textContent = String(bookmarks.length);
  }
  return bookmarks;
}

function formatBookmarkTime(timestamp) {
  const date = new Date(Number(timestamp));
  if (!Number.isFinite(date.getTime())) return '';
  return date.toLocaleString('vi-VN', { dateStyle: 'short', timeStyle: 'short' });
}

async function handleBookmarkNavigation(bookmarkId) {
  if (bookmarkNavigationPending || searchNavigationPending) return;
  if (scrollRestoreTimeout) {
    clearTimeout(scrollRestoreTimeout);
    scrollRestoreTimeout = null;
  }
  bookmarkNavigationPending = true;

  try {
    const bookmark = await getBookmark(bookmarkId);
    const numericValue = value => (typeof value === 'number'
      || (typeof value === 'string' && value.trim() !== '')) ? Number(value) : NaN;
    const bookId = numericValue(bookmark?.bookId);
    const chapterIndex = numericValue(bookmark?.chapterIndex);
    const paragraphIndex = numericValue(bookmark?.paragraphIndex);
    if (!bookmark || !Number.isSafeInteger(bookId) || bookId <= 0
      || !Number.isSafeInteger(chapterIndex) || chapterIndex < 0
      || !Number.isSafeInteger(paragraphIndex) || paragraphIndex < 0) {
      showToast('Bookmark này không còn khả dụng.', 'warning');
      return;
    }

    const book = await getBook(bookId);
    if (!book) {
      showToast('Tác phẩm của bookmark này đã bị xóa.', 'warning');
      return;
    }
    const chapters = await getBookChaptersMeta(bookId);
    const chapterExists = chapters.some(chapter => Number(chapter.chapterIndex) === chapterIndex);
    if (!chapterExists) {
      showToast('Không tìm thấy chương của bookmark này.', 'warning');
      return;
    }

    const alreadyAtTarget = DOM.readerView.classList.contains('active')
      && Number(state.currentBook?.id) === bookId
      && Number(state.currentChapterIndex) === chapterIndex
      && Number(state.currentChapter?.chapterIndex) === chapterIndex;

    closeTocDrawer();
    let loaded = true;
    if (!alreadyAtTarget) {
      if (DOM.readerView.classList.contains('active') && Number(state.currentBook?.id) === bookId) {
        loaded = await loadChapter(chapterIndex, 0, 0);
      } else {
        const savedChapterMatches = Number(book.currentChapterIndex) === chapterIndex;
        loaded = await openBook(bookId, chapterIndex, {
          preserveChunk: savedChapterMatches,
          restoreScroll: false
        });
      }
    }
    if (!loaded || !DOM.readerView.classList.contains('active')
      || Number(state.currentBook?.id) !== bookId
      || Number(state.currentChapterIndex) !== chapterIndex
      || Number(state.currentChapter?.chapterIndex) !== chapterIndex) {
      return;
    }

    const paragraphs = [...DOM.readerBodyContent.querySelectorAll('.reader-paragraph')];
    const resolved = resolveBookmarkAnchor(bookmark, paragraphs);
    if (!resolved?.paragraph) {
      showToast('Không thể xác định chính xác vị trí cũ. Đã mở chương tương ứng.', 'info');
      return;
    }

    clearBookmarkFocus();
    if (alreadyAtTarget && state.autoScroll?.isActive) {
      state.autoScroll.pauseTemporary();
    }
    readerProgressRebasePending = true;
    resolved.paragraph.scrollIntoView({ behavior: 'smooth', block: 'center', inline: 'nearest' });
    focusBookmarkParagraph(resolved.paragraph);
    resetProgressBaselineAtParagraph(resolved.paragraph);
    noteTrustedKeyboardActivity();
  } catch (error) {
    console.warn('Bookmark navigation failed:', error);
    showToast('Không thể mở bookmark lúc này.', 'warning');
  } finally {
    bookmarkNavigationPending = false;
  }
}

function clearBookmarkFocus() {
  if (bookmarkFocusTimer !== null) {
    clearTimeout(bookmarkFocusTimer);
    bookmarkFocusTimer = null;
  }
  if (activeBookmarkFocus) {
    const { element, hadTabIndex, previousTabIndex } = activeBookmarkFocus;
    element.classList.remove('bookmark-focus');
    if (hadTabIndex) element.setAttribute('tabindex', previousTabIndex);
    else element.removeAttribute('tabindex');
    activeBookmarkFocus = null;
  }
}

function focusBookmarkParagraph(paragraph) {
  const hadTabIndex = paragraph.hasAttribute('tabindex');
  const previousTabIndex = paragraph.getAttribute('tabindex');
  activeBookmarkFocus = { element: paragraph, hadTabIndex, previousTabIndex };
  paragraph.setAttribute('tabindex', '-1');
  paragraph.classList.add('bookmark-focus');
  try { paragraph.focus({ preventScroll: true }); }
  catch { paragraph.focus(); }

  // This timeout only removes temporary visual focus; chapter rendering is awaited through loadChapter/openBook.
  bookmarkFocusTimer = window.setTimeout(clearBookmarkFocus, 2200);
}

  const list = bookmarks || await getBookBookmarks(bookId);

  DOM.bookmarksCardsList.innerHTML = list.map(bookmark => {

    const chapterTitle = chapter?.title || `Chương ${Number(bookmark.chapterIndex) + 1}`;
    const label = typeof bookmark.label === 'string' ? bookmark.label : '';
    const time = formatBookmarkTime(bookmark.createdAt);
    return `

          <span class="bookmark-card-chapter">${escapeHTML(chapterTitle)}</span>

          <span class="bookmark-card-label" ${label ? '' : 'hidden'}>🏷️ ${escapeHTML(label)}</span>
          <button class="bookmark-label-toggle" type="button" aria-label="${label ? 'Sửa nhãn bookmark' : 'Thêm nhãn bookmark'}" title="${label ? 'Sửa nhãn' : 'Thêm nhãn'}">${label ? 'Sửa nhãn' : 'Thêm nhãn'}</button>
        </div>
        <form class="bookmark-label-form" hidden>
          <label class="sr-only" for="bookmark-label-${Number(bookmark.id)}">Nhãn ngắn</label>
          <input id="bookmark-label-${Number(bookmark.id)}" class="bookmark-label-input" type="text" maxlength="120" placeholder="Nhãn ngắn (không bắt buộc)" value="${escapeHTML(label)}">
          <button class="btn-primary btn-mini" type="submit">Lưu</button>
          <button class="bookmark-label-cancel" type="button">Hủy</button>

    const form = card.querySelector('.bookmark-label-form');
    const input = card.querySelector('.bookmark-label-input');
    const editButton = card.querySelector('.bookmark-label-toggle');
    editButton.addEventListener('click', () => {
      form.hidden = false;
      editButton.hidden = true;
      input.focus();
    });
    card.querySelector('.bookmark-label-cancel').addEventListener('click', () => {
      form.hidden = true;
      editButton.hidden = false;
      input.value = card.querySelector('.bookmark-card-label').textContent.replace(/^🏷️\s*/, '');
    });
    form.addEventListener('submit', async event => {
      event.preventDefault();
      const updated = await updateBookmark(bookmarkId, { label: input.value });
      if (!updated) {
        showToast('Không thể cập nhật nhãn bookmark.', 'error');
        return;
      }
      showToast(input.value.trim() ? 'Đã lưu nhãn bookmark.' : 'Đã bỏ nhãn bookmark.', 'success');
      await renderBookmarkHubList();
    });
    card.querySelector('.bookmark-jump-button').addEventListener('click', () => {
      void handleBookmarkNavigation(bookmarkId);
    });
    card.querySelector('.bookmark-delete-button').addEventListener('click', async () => {
      const deleted = await deleteBookmark(bookmarkId);
      if (!deleted) {
        showToast('Không thể xóa bookmark lúc này.', 'error');
        return;
      }
      showToast('Đã xóa bookmark.', 'info');
      await refreshBookmarksCount(bookId);
      await renderBookmarkHubList();
    });
  });

/** Switches between the existing TOC, Quotes, and Bookmark drawer panels. */
function switchDrawerTab(tabName) {
  const tabs = [
    { name: 'toc', button: DOM.tabBtnToc, view: DOM.tabViewToc },
    { name: 'quotes', button: DOM.tabBtnQuotes, view: DOM.tabViewQuotes },
    { name: 'bookmarks', button: DOM.tabBtnBookmarks, view: DOM.tabViewBookmarks }
  ];
  const selected = tabs.some(tab => tab.name === tabName) ? tabName : 'toc';
  for (const tab of tabs) {
    const active = tab.name === selected;
    tab.button?.classList.toggle('active', active);
    tab.button?.setAttribute('aria-selected', String(active));
    tab.button?.setAttribute('tabindex', active ? '0' : '-1');
    if (tab.view) tab.view.style.display = active ? 'flex' : 'none';
  }

  if (selected === 'quotes') void renderQuotesHubList();
  if (selected === 'bookmarks') void renderBookmarkHubList();
}

function handleDrawerTabKeydown(event, currentTab) {
  const tabNames = ['toc', 'quotes', 'bookmarks'];
  const currentIndex = tabNames.indexOf(currentTab);
  let nextIndex = currentIndex;
  if (event.key === 'ArrowRight' || event.key === 'ArrowDown') nextIndex = (currentIndex + 1) % tabNames.length;
  else if (event.key === 'ArrowLeft' || event.key === 'ArrowUp') nextIndex = (currentIndex - 1 + tabNames.length) % tabNames.length;
  else if (event.key === 'Home') nextIndex = 0;
  else if (event.key === 'End') nextIndex = tabNames.length - 1;
  else return;

  event.preventDefault();
  const nextTab = tabNames[nextIndex];
  const nextButton = {
    toc: DOM.tabBtnToc,
    quotes: DOM.tabBtnQuotes,
    bookmarks: DOM.tabBtnBookmarks
  }[nextTab];
  nextButton?.focus();
  switchDrawerTab(nextTab);
}

/**

      const timeStr = q.createdAt ? new Date(q.createdAt).toLocaleDateString('vi-VN') : '';
      const colorClass = `border-${q.color || 'yellow'}`;

      html += `
        <div class="quote-card ${colorClass}" data-quote-id="${q.id}" data-chapter-index="${q.chapterIndex}">
          <div class="quote-card-header">
            <span class="quote-card-chapter">${escapeHTML(chTitle)}</span>
            <span>${timeStr}</span>
          </div>

              📍 Đến đoạn này
            </button>
            <button class="btn-secondary btn-mini btn-quote-copy" data-quote-id="${q.id}" title="Sao chép trích dẫn">
              📋 Chép
            </button>
            <button class="btn-secondary btn-mini btn-quote-delete" data-quote-id="${q.id}" title="Xóa trích dẫn">
              🗑️
            </button>
          </div>
        </div>
      `;
    }

    DOM.quotesCardsList.innerHTML = html;

    // Attach card event listeners
    DOM.quotesCardsList.querySelectorAll('.btn-quote-jump').forEach(btn => {
      btn.addEventListener('click', async (e) => {
        e.stopPropagation();
        const chIdx = parseInt(btn.getAttribute('data-chapter-index'), 10);
        const qId = btn.getAttribute('data-quote-id');
        await jumpToHighlight(chIdx, qId);
      });
    });

    DOM.quotesCardsList.querySelectorAll('.btn-quote-copy').forEach(btn => {
      btn.addEventListener('click', async (e) => {
        e.stopPropagation();
        const qId = btn.getAttribute('data-quote-id');
        const quote = quotes.find(q => Number(q.id) === Number(qId));
        if (quote) {
          const meta = state.chaptersMeta.find(c => c.chapterIndex === quote.chapterIndex);
          const chTitle = meta ? meta.title : `Chương ${quote.chapterIndex + 1}`;
          const formatted = formatQuoteForClipboard(quote.selectedText, state.currentBook.title, chTitle);
          try {
            await navigator.clipboard.writeText(formatted);
            showToast('Đã sao chép trích dẫn vào clipboard!', 'success');
          } catch (err) {
            showToast('Không thể sao chép trích dẫn: ' + (err?.message || String(err)), 'error');
          }
        }
      });
    });

    DOM.quotesCardsList.querySelectorAll('.btn-quote-delete').forEach(btn => {
      btn.addEventListener('click', async (e) => {
        e.stopPropagation();
        const qId = btn.getAttribute('data-quote-id');

          showToast('Đã xóa trích dẫn.', 'info');
          await renderQuotesHubList();
          await loadAndApplyChapterHighlights();
        }
      });
    });

    // Whole card click jumps to quote
    DOM.quotesCardsList.querySelectorAll('.quote-card').forEach(card => {
      card.addEventListener('click', async () => {
        const chIdx = parseInt(card.getAttribute('data-chapter-index'), 10);
        const qId = card.getAttribute('data-quote-id');
        await jumpToHighlight(chIdx, qId);
      });
    });

  } catch (err) {
    console.error('Lỗi hiển thị danh sách trích dẫn:', err);
  }
}

/**
 * Jumps to a specific highlight: switches chapter if needed, scrolls into view smoothly, and highlights.
 * @param {number} targetChapterIndex
 * @param {string|number} highlightId
 */
async function jumpToHighlight(targetChapterIndex, highlightId) {
  closeTocDrawer();

  if (targetChapterIndex !== state.currentChapterIndex) {
    await loadChapter(targetChapterIndex);
  }

  // Small delay to ensure rendering finished
  setTimeout(() => {
    const mark = DOM.readerBodyContent.querySelector(`mark.reader-highlight[data-highlight-id="${highlightId}"]`);
    if (mark) {
      mark.scrollIntoView({ behavior: 'smooth', block: 'center' });
      mark.style.outline = '3px solid var(--primary-color)';
      setTimeout(() => {
        mark.style.outline = '';
      }, 2000);
    } else {
      showToast('Đã mở chương chứa đoạn trích!', 'info');
    }
  }, 200);
}

/**
 * Exports all quotes and notes for current book as Markdown file for Obsidian / Notion.
 */
async function handleExportQuotesMarkdown() {
  if (!state.currentBook) return;

  try {
    const quotes = await getBookHighlights(state.currentBook.id);
    const md = exportHighlightsToMarkdown(state.currentBook.title, state.currentBook.author, quotes, state.chaptersMeta);

    const safeTitle = (state.currentBook.title || 'trich-dan')
      .toLowerCase()
      .replace(/[^\w\s-]/g, '')
      .replace(/\s+/g, '-');
    const fileName = `${safeTitle}-trich-dan.md`;

    const blob = new Blob([md], { type: 'text/markdown;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = fileName;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);

    showToast(`Đã xuất ${quotes.length} trích dẫn ra file Markdown!`, 'success');
  } catch (err) {
    showToast('Lỗi xuất Markdown: ' + err.message, 'error');
  }
}

/**
 * Renders custom rules into the dictionary table.
 */
async function renderDictionaryRulesTable() {
  if (!DOM.dictRulesTbody) return;

  const currentBookId = state.currentBook ? state.currentBook.id : null;
  const rules = await getCustomRules(currentBookId);
  DOM.dictRulesCount.textContent = rules.length;

  if (rules.length === 0) {
    DOM.dictRulesTbody.innerHTML = '';
    DOM.dictEmptyState.style.display = 'block';
    return;
  }

  DOM.dictEmptyState.style.display = 'none';

  DOM.dictRulesTbody.innerHTML = rules.map(r => {
    const isBookSpecific = r.bookId !== null && r.bookId !== undefined && r.bookId !== 'global';
    const scopeLabel = isBookSpecific ? 'Truyện này' : 'Toàn hệ thống';

    return `
      <tr data-rule-id="${r.id}">
        <td><strong>${escapeHTML(r.findText)}</strong></td>
        <td>${r.replaceText ? escapeHTML(r.replaceText) : '<span style="color: var(--text-muted); font-style: italic;">[Xóa bỏ]</span>'}</td>
        <td>${r.isCaseSensitive ? '✓ Có' : 'Không'}</td>
        <td>${r.isRegex ? '<span class="badge-pill-mini">Regex</span>' : 'Thường'}</td>
        <td><small style="color: var(--text-muted);">${scopeLabel}</small></td>
        <td>
          <input type="checkbox" class="dict-rule-toggle" data-rule-id="${r.id}" ${r.isEnabled !== false ? 'checked' : ''} title="Bật/Tắt quy tắc">
        </td>
        <td>
          <button class="btn-icon btn-rule-delete" data-rule-id="${r.id}" title="Xóa quy tắc này" style="color: var(--danger-color); padding: 2px;">
            <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
              <polyline points="3 6 5 6 21 6"/>
              <path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"/>
            </svg>
          </button>
        </td>
      </tr>
    `;
  }).join('');

  // Bind toggle & delete events
  DOM.dictRulesTbody.querySelectorAll('.dict-rule-toggle').forEach(chk => {
    chk.addEventListener('change', async () => {
      const id = chk.getAttribute('data-rule-id');
      await updateCustomRule(Number(id), { isEnabled: chk.checked });
      await reloadCurrentChapterText();
      showToast(chk.checked ? 'Đã kích hoạt quy tắc' : 'Đã tạm tắt quy tắc', 'info');
    });
  });

  DOM.dictRulesTbody.querySelectorAll('.btn-rule-delete').forEach(btn => {
    btn.addEventListener('click', async () => {
      const id = btn.getAttribute('data-rule-id');
      await deleteCustomRule(Number(id));
      showToast('Đã xóa quy tắc thay thế.', 'info');
      await renderDictionaryRulesTable();
      await reloadCurrentChapterText();
    });
  });
}

/**
 * Handles adding a new custom dictionary replacement rule.
 */
async function handleAddDictionaryRule() {
  const findText = DOM.dictFindInput.value.trim();
  const replaceText = DOM.dictReplaceInput.value;
  const isCaseSensitive = DOM.dictCaseChk.checked;
  const isRegex = DOM.dictRegexChk.checked;
  const onlyThisBook = DOM.dictScopeChk.checked;

  if (!findText) {
    showToast('Vui lòng nhập từ gốc cần tìm (Find)!', 'warning');
    DOM.dictFindInput.focus();
    return;
  }

  // Validate regex syntax if regex enabled
  if (isRegex) {
    try {
      new RegExp(findText);
    } catch (err) {
      showToast('Biểu thức chính quy (Regex) không hợp lệ: ' + err.message, 'error');
      DOM.dictFindInput.focus();
      return;
    }
  }

  const bookId = onlyThisBook && state.currentBook ? state.currentBook.id : null;

  try {
    await addCustomRule({
      bookId,
      findText,
      replaceText,
      isCaseSensitive,
      isRegex,
      isEnabled: true
    });

    DOM.dictFindInput.value = '';
    DOM.dictReplaceInput.value = '';
    DOM.dictFindInput.focus();

    showToast('Đã thêm quy tắc mới!', 'success');
    await renderDictionaryRulesTable();
    await reloadCurrentChapterText();

  } catch (err) {
    showToast('Lỗi thêm quy tắc: ' + err.message, 'error');
  }
}

/**
 * Imports dictionary rules from JSON file.
 */
async function handleImportRulesJSON(e) {
  const file = e.target.files && e.target.files[0];
  if (!file) return;

  try {
    const text = await file.text();
    const data = JSON.parse(text);
    const validated = validateRules(data);

    const targetBookId = DOM.dictScopeChk.checked && state.currentBook ? state.currentBook.id : null;
    const count = await importCustomRulesJSON(validated, targetBookId, false);

    showToast(`Đã nhập thành công ${count} quy tắc thay thế!`, 'success');
    await renderDictionaryRulesTable();
    await reloadCurrentChapterText();

  } catch (err) {
    showToast('Lỗi nhập file từ điển: ' + err.message, 'error');
  } finally {
    DOM.dictImportFileInput.value = '';
  }
}

/**
 * Re-cleans current chapter text through the updated pipeline and updates DOM & TTS
 * without resetting user's reading position.
 */
async function reloadCurrentChapterText() {
  if (!state.currentBook || !state.currentChapter) return;

  try {
    const currentScrollY = window.scrollY;

    // Fetch refreshed rules
    const rules = await getCustomRules(state.currentBook.id);
    state.customRules = rules;

    const cleanedContent = cleanChapterText(state.currentChapter.content, {
      rules,
      enableCleaner: state.enableGlobalCleaner
    });

    // Re-render DOM
    renderChapterContent(cleanedContent);

    // Re-load text into TTS engine
    state.tts.loadText(cleanedContent);

    // Re-apply highlights
    await loadAndApplyChapterHighlights();

    // Preserve scroll position
    window.scrollTo({ top: currentScrollY, behavior: 'instant' });

  } catch (err) {
    console.error('Lỗi nạp lại văn bản đã lọc:', err);
  }
}

/**
 * Initializes Sleep Timer & Gradual Volume Fade-out.
 */
function initSleepTimer() {
  state.sleepTimer = new SleepTimer({
    fadeDuration: 30,
    onTick: (status) => {
      updateSleepTimerUI(status);
    },
    onFade: (fadeRatio) => {
      // Fade TTS volume from 1.0 to 0
      state.tts?.setVolume(fadeRatio);
      // Fade ambient sound volume
      state.ambientMixer?.setFadeFactor(fadeRatio);
    },
    onExpire: ({ reason }) => {
      handleSleepTimerExpire(reason);
    },
    onCancel: () => {
      const restoredTtsVol = state.sleepTimer?.baseTtsVolume ?? 1.0;
      state.tts?.setVolume(restoredTtsVol);
      const restoredAmbientFactor = state.sleepTimer?.baseAmbientVolume ?? 1.0;
      state.ambientMixer?.setFadeFactor(restoredAmbientFactor);
      updateSleepTimerUI({ isActive: false });
    }
  });

  // Dock button click -> Open modal
  if (DOM.btnDockSleepTimer) {
    DOM.btnDockSleepTimer.addEventListener('click', () => {
      openSleepTimerModal();
    });
  }

  // Modal close buttons
  if (DOM.btnCloseSleepTimer) {
    DOM.btnCloseSleepTimer.addEventListener('click', closeSleepTimerModal);
  }
  if (DOM.btnCloseTimerModal) {
    DOM.btnCloseTimerModal.addEventListener('click', closeSleepTimerModal);
  }
  if (DOM.sleepTimerModal) {
    DOM.sleepTimerModal.addEventListener('click', (e) => {
      if (e.target === DOM.sleepTimerModal) closeSleepTimerModal();
    });
  }

  // Presets buttons (15m, 30m, 45m, 60m)
  document.querySelectorAll('.timer-preset-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const minutes = parseInt(btn.getAttribute('data-minutes'), 10);
      if (minutes && state.sleepTimer) {
        state.sleepTimer.start(minutes);
        showToast(`Đã hẹn giờ tắt sau ${minutes} phút (kèm Fade-out 30s)`, 'success');
        updatePresetButtonActive(minutes);
        closeSleepTimerModal();
      }
    });
  });

  // Chapter end button
  if (DOM.btnTimerChapterEnd) {
    DOM.btnTimerChapterEnd.addEventListener('click', () => {
      if (state.sleepTimer) {
        state.sleepTimer.setChapterEndMode();
        showToast('Đã bật: Tự động dừng sau khi đọc xong chương hiện tại', 'success');
        updatePresetButtonActive('chapter_end');
        closeSleepTimerModal();
      }
    });
  }

  // Cancel timer button
  if (DOM.btnCancelSleepTimer) {
    DOM.btnCancelSleepTimer.addEventListener('click', () => {
      if (state.sleepTimer) {
        state.sleepTimer.cancel();
        showToast('Đã hủy hẹn giờ ngủ', 'info');
        updatePresetButtonActive(null);
      }
    });
  }
}

/**
 * Updates Sleep Timer badge on Floating Dock and modal countdown display.
 */
function updateSleepTimerUI(status) {
  if (!DOM.btnDockSleepTimer || !DOM.dockSleepTimerBadge) return;

  if (status && status.isActive) {
    DOM.btnDockSleepTimer.classList.add('active');
    DOM.dockSleepTimerBadge.style.display = 'inline-block';
    DOM.dockSleepTimerBadge.textContent = status.formatted || '';
    DOM.dockSleepTimerBadge.classList.toggle('fading', !!status.isFading);

    if (DOM.sleepTimerStatusCard) {
      DOM.sleepTimerStatusCard.style.display = 'block';
    }
    if (DOM.timerCountdownVal) {
      DOM.timerCountdownVal.textContent = status.formatted || '';
    }
    if (DOM.timerStatusDesc) {
      if (status.mode === 'chapter_end') {
        DOM.timerStatusDesc.textContent = 'Đang bật: Dừng khi hoàn thành câu cuối của chương này.';
      } else if (status.isFading) {
        DOM.timerStatusDesc.textContent = 'Đang giảm dần âm lượng (Fade-out 30s)...';
      } else {
        DOM.timerStatusDesc.textContent = 'Đang đếm ngược... Tự động giảm dần âm lượng trong 30 giây cuối.';
      }
    }
  } else {
    DOM.btnDockSleepTimer.classList.remove('active');
    DOM.dockSleepTimerBadge.style.display = 'none';
    DOM.dockSleepTimerBadge.classList.remove('fading');
    if (DOM.sleepTimerStatusCard) {
      DOM.sleepTimerStatusCard.style.display = 'none';
    }
    updatePresetButtonActive(null);
  }
}

  document.querySelectorAll('.timer-preset-btn').forEach(btn => {

/**
 * Handles sleep timer expiration.
 */
function handleSleepTimerExpire(reason) {
  // 1. Stop TTS and restore base volume
  if (state.tts) {
    state.tts.stop();
    const restoredTtsVol = state.sleepTimer?.baseTtsVolume ?? 1.0;
    state.tts.setVolume(restoredTtsVol);
  }

  // 2. Stop/pause ambient sound & restore base volume factor
  if (state.ambientMixer) {
    state.ambientMixer.pauseAll();
    const restoredAmbientFactor = state.sleepTimer?.baseAmbientVolume ?? 1.0;
    state.ambientMixer.setFadeFactor(restoredAmbientFactor);
  }

  // 3. Stop background silent audio
  if (state.mediaSession) {
    state.mediaSession.stopBackgroundSession();
  }

  // 4. Update UI
  updateSleepTimerUI({ isActive: false });

  const msg = reason === 'chapter_end'
    ? 'Đã đọc hết chương. Hẹn giờ ngủ đã tạm dừng giọng đọc và âm thanh.'
    : 'Đã hết thời gian hẹn giờ ngủ. Chúc bạn ngủ ngon!';
  showToast(msg, 'info');
}

/**

  state.ambientMixer = new AmbientSoundMixer();

  // Load saved state from storage

    state.ambientMixer.loadState(savedMixerState, false);

      const masterPct = Math.round(savedMixerState.masterVolume * 100);

      if (DOM.labelAmbientMasterVal) DOM.labelAmbientMasterVal.textContent = `${masterPct}%`;

  if (DOM.btnDockAmbient) {
    DOM.btnDockAmbient.addEventListener('click', () => {
      openAmbientMixerModal();
    });

  if (DOM.btnCloseAmbientModal) {
    DOM.btnCloseAmbientModal.addEventListener('click', closeAmbientMixerModal);
  }
  if (DOM.btnCloseAmbientDone) {
    DOM.btnCloseAmbientDone.addEventListener('click', closeAmbientMixerModal);
  }
  if (DOM.ambientMixerModal) {
    DOM.ambientMixerModal.addEventListener('click', (e) => {
      if (e.target === DOM.ambientMixerModal) closeAmbientMixerModal();
    });
  }

  // Master volume slider
  if (DOM.sliderAmbientMaster) {
    DOM.sliderAmbientMaster.addEventListener('input', (e) => {
      const val = parseInt(e.target.value, 10) / 100;
      state.ambientMixer.setMasterVolume(val);
      if (DOM.labelAmbientMasterVal) {
        DOM.labelAmbientMasterVal.textContent = `${e.target.value}%`;
      }
      saveAmbientMixerState();
    });
  }

  // Stop All button
  if (DOM.btnAmbientStopAll) {
    DOM.btnAmbientStopAll.addEventListener('click', () => {
      state.ambientMixer.stopAll();
      renderAmbientTracks();
      updateAmbientDockIndicator();
      saveAmbientMixerState();

    });
  }
}

/**

 */
function renderAmbientTracks() {
  if (!DOM.ambientTracksContainer || !state.ambientMixer) return;

  DOM.ambientTracksContainer.innerHTML = AMBIENT_SOUND_PRESETS.map(preset => {
    const track = state.ambientMixer.tracks.get(preset.id);
    const isPlaying = track ? track.isPlaying : false;
    const volPercent = track ? Math.round(track.volume * 100) : Math.round(preset.defaultVolume * 100);

    return `
      <div class="ambient-track-card ${isPlaying ? 'active' : ''}" data-sound-id="${preset.id}">
        <div class="ambient-track-left">
          <div class="ambient-track-icon">${preset.icon}</div>
          <div class="ambient-track-info">
            <div class="ambient-track-title">${escapeHTML(preset.name)}</div>
            <div class="ambient-track-desc">${escapeHTML(preset.description)}</div>
          </div>
        </div>
        <div class="ambient-track-right">
          <div class="ambient-track-slider-group" title="Âm lượng tiếng ${escapeHTML(preset.name)}">
            <input type="range" class="form-range ambient-track-slider" min="0" max="100" value="${volPercent}" data-sound-id="${preset.id}">
            <span class="ambient-track-vol-val">${volPercent}%</span>
          </div>
          <label class="switch-toggle" title="Bật/Tắt âm thanh này">
            <input type="checkbox" class="ambient-track-toggle" data-sound-id="${preset.id}" ${isPlaying ? 'checked' : ''}>
            <span class="switch-slider"></span>
          </label>
        </div>
      </div>
    `;
  }).join('');

  // Attach track toggle listeners
  DOM.ambientTracksContainer.querySelectorAll('.ambient-track-toggle').forEach(chk => {
    chk.addEventListener('change', (e) => {
      const soundId = chk.getAttribute('data-sound-id');
      const isChecked = chk.checked;
      state.ambientMixer.toggleSound(soundId, isChecked);

      const card = chk.closest('.ambient-track-card');
      if (card) card.classList.toggle('active', isChecked);

      updateAmbientDockIndicator();
      saveAmbientMixerState();
    });
  });

  // Attach track volume slider listeners
  DOM.ambientTracksContainer.querySelectorAll('.ambient-track-slider').forEach(slider => {
    slider.addEventListener('input', (e) => {
      const soundId = slider.getAttribute('data-sound-id');
      const volumeFloat = parseInt(e.target.value, 10) / 100;
      state.ambientMixer.setSoundVolume(soundId, volumeFloat);

      const valLabel = slider.parentElement?.querySelector('.ambient-track-vol-val');
      if (valLabel) valLabel.textContent = `${e.target.value}%`;

      saveAmbientMixerState();
    });
  });
}

/**
 * Initializes MediaSession API with Bluetooth headset controls and silent audio background support.
 */
function initMediaSession() {
  if (state.mediaSession) {
    try {
      state.mediaSession.destroy();
    } catch {}
    state.mediaSession = null;
  }
  state.mediaSession = new MediaSessionHelper({
    onPlay: () => {
      if (state.tts && !state.tts.isPlaying) {
        if (state.tts.isPaused) {
          state.tts.resume();
        } else {
          state.tts.jumpToIndex(state.tts.currentIndex || 0, true);
        }
      }
    },
    onPause: () => {
      if (state.tts && state.tts.isPlaying) {
        state.tts.pause();
      }
    },
    onPrevTrack: () => {
      if (state.tts && state.tts.currentIndex > 0) {
        state.tts.prev();
      } else if (state.currentChapterIndex > 0) {
        navigateChapter(-1);
      }
    },
    onNextTrack: () => {
      if (state.tts && state.tts.currentIndex < state.tts.chunks.length - 1) {
        state.tts.next();
      } else if (state.currentChapterIndex < state.chaptersMeta.length - 1) {
        navigateChapter(1);
      }
    },
    onSeekBackward: (offset) => {
      if (state.tts) {
        const target = Math.max(0, state.tts.currentIndex - 2);
        state.tts.jumpToIndex(target, state.tts.isPlaying);
      }
    },
    onSeekForward: (offset) => {
      if (state.tts) {
        const target = Math.min(state.tts.chunks.length - 1, state.tts.currentIndex + 2);
        state.tts.jumpToIndex(target, state.tts.isPlaying);
      }
    },
    onStop: () => {
      if (state.tts) {
        state.tts.stop();
      }
    }
  });
}
