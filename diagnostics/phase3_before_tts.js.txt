/**
 * Returns a human-friendly decorated name for a browser Web Speech voice.
 */
export function formatBrowserVoiceLabel(voice) {
  const name = voice.name || '';
  if (/hoaimy/i.test(name)) {
    return '👩 Hoài My (Nữ - Bắc, Tự nhiên - Edge Natural)';
  }
  if (/namminh|microsoft nam/i.test(name)) {
    return '👨 Nam (Nam - Bắc, Trầm ấm - Edge Natural)';
  }
  if (/google\s*ti[ếe]ng\s*vi[ệe]t/i.test(name)) {
    return '👩 Google Tiếng Việt (Nữ chuẩn - Chrome/Android)';
  }
  if (/microsoft an/i.test(name)) {
    return '👩 An (Nữ - Tự nhiên - Windows)';
  }
  if (/microsoft mai/i.test(name)) {
    return '👩 Mai (Nữ - Tiếng Việt - Windows)';
  }
  if (/linh/i.test(name)) {
    return '👩 Linh (Nữ - Tiếng Việt - Apple)';
  }
  // Generic fallback format
  return `🇻🇳 ${name}`;
}

/**
 * Generates a simple hash string for text caching.
 */
function hashString(str) {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = ((hash << 5) - hash) + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash).toString(36);
}

  /**
   * Initializes HTMLAudioElement events for Cloud TTS streaming.
   */
  initAudioElement() {
    if (this.disposed || !this.audioElement || this.audioEventListeners.length > 0) return;
    const listen = (type, handler) => {
      this.audioEventListeners.push({ type, handler });
      this.audioElement.addEventListener(type, handler);
    };

    listen('playing', () => {
      if (this.activeCloudRequestId === null
        || this.activeCloudRequestId !== this.playbackRequestId) return;
      if (this.lastChunkEndedTimestamp) {
        const gapMs = Math.max(0, Date.now() - this.lastChunkEndedTimestamp);
        if (this.queueScheduler) {
          this.queueScheduler.recordChunkGap(gapMs);
        }
        this.lastChunkEndedTimestamp = null;
      }
      if (this.cloudChunkStartedIndex !== this.currentIndex) {
        this.cloudChunkStartedIndex = this.currentIndex;
        this.emitPlaybackCallback(this.onActualChunkStart, this.currentIndex);
        if (this.queueScheduler && this.chunks && this.chunks.length > 0) {
          this.queueScheduler.markPlaying(this.currentIndex);
          this.queueScheduler.updateWindow({
            currentIndex: this.currentIndex,
            chunks: this.chunks,
            config: this.cloudConfig,
            rate: this.rate,
            generation: this.generationToken
          });
        }
      }
      this.emitActualPlayback(true);
    });
    listen('pause', () => {
      if (this.activeCloudRequestId === this.playbackRequestId) {
        this.emitActualPlayback(false);
      }
    });

    listen('ended', () => {
      if (this.activeCloudRequestId === null
        || this.activeCloudRequestId !== this.playbackRequestId) return;
      this.activeCloudRequestId = null;
      this.lastChunkEndedTimestamp = Date.now();
      const completedIndex = this.currentIndex;
      if (this.queueScheduler) {
        this.queueScheduler.markConsumed(completedIndex);
      }
      const currentGen = this.generationToken;
      const continuesPlayback = this.isPlaying && !this.isPaused && completedIndex + 1 < this.chunks.length;
      if (!continuesPlayback) this.emitActualPlayback(false);
      this.emitPlaybackCallback(this.onActualChunkComplete, completedIndex);
      this.cloudChunkStartedIndex = null;
      this.cleanupCurrentAudioUrl();
      if (typeof this.onChunkEnd === 'function') {
        this.onChunkEnd(completedIndex);
      }

      if (this.isPlaying && !this.isPaused && this.generationToken === currentGen) {
        if (this.checkChunkVelocityCircuitBreaker()) return;

        if (completedIndex + 1 < this.chunks.length) {
          const nextIndex = completedIndex + 1;
          const advanceToken = ++this.advanceSequenceToken;
          this.isAdvancingChunk = true;
          if (this.transitionTimer) clearTimeout(this.transitionTimer);
          this.transitionTimer = setTimeout(() => {
            this.transitionTimer = null;
            this.isAdvancingChunk = false;
            if (this.generationToken === currentGen && advanceToken === this.advanceSequenceToken && this.isPlaying && !this.isPaused) {
              this.currentIndex = nextIndex;
              this.speakCurrentChunk();
            }
          }, 0);
        } else {
          this.emitPlaybackCallback(this.onNaturalComplete, completedIndex);
          this.stop();
          this.emitStatus('completed');
        }
      }
    });

    listen('error', (e) => {
      if (this.activeCloudRequestId === null
        || this.activeCloudRequestId !== this.playbackRequestId) return;
      this.emitActualPlayback(false);
      this.cloudChunkStartedIndex = null;
      this.activeCloudRequestId = null;
      this.pendingCloudRequestId = null;
      console.warn('Lỗi phát Cloud Audio:', e);
      this.cleanupCurrentAudioUrl();

      // Graceful fallback to browser speech for current chunk
      if (this.queueScheduler) {
        this.queueScheduler.invalidate(this.generationToken);
      }
      if (typeof this.onFallbackWarning === 'function') {
        this.onFallbackWarning('Cloud Audio bị lỗi. Tự động chuyển sang giọng trình duyệt.');
      }
      this.engine = 'browser';
      this.speakBrowserChunk(this.chunks[this.currentIndex], this.playbackRequestId);
    });
  }

  emitActualPlayback(isPlaying) {
    const next = Boolean(isPlaying);
    if (this.actualPlaybackActive === next) return;
    this.actualPlaybackActive = next;
    const ts = typeof this.now === 'function' ? this.now() : Date.now();
    this.emitPlaybackCallback(this.onPlaybackStateChange, next, ts);
  }

  emitPlaybackCallback(callback, ...args) {
    if (typeof callback !== 'function') return;
    try { callback(...args); }
    catch (error) { console.warn('TTS playback observer failed; playback continues.', error); }
  }

  /**
   * Circuit Breaker cấp độ chunk: Phát hiện và dập tắt tức thì callback storm hoặc chuỗi onend lặp bất thường.
   * Phân biệt giữa hành động chủ động của người dùng (Next/Seek) và vòng lặp tự động.
   * @returns {boolean} true nếu cầu chì bị ngắt (trip)
   */
  checkChunkVelocityCircuitBreaker() {
    const now = Date.now();
    this.autoChunkTransitionTimestamps = (this.autoChunkTransitionTimestamps || []).filter(t => now - t < 1000);
    this.autoChunkTransitionTimestamps.push(now);

    const MAX_AUTO_CHUNKS_PER_SEC = 60;
    if (this.autoChunkTransitionTimestamps.length >= MAX_AUTO_CHUNKS_PER_SEC) {
      console.error(`>>> [VietReader] Chunk Velocity Circuit Breaker: Tự động dừng phát do callback storm quá nhanh (${this.autoChunkTransitionTimestamps.length} chunks / 1000ms).`);
      this.generationToken++;
      this.advanceSequenceToken++;
      this.isAdvancingChunk = false;
      this.stop();
      this.emitStatus('error');
      if (typeof this.onFallbackWarning === 'function') {
        this.onFallbackWarning('Đã tự động dừng phát do phát hiện vòng lặp giọng đọc bất thường để bảo vệ tài nguyên trình duyệt.');
      }
      return true;
    }
    return false;
  }

  /**
   * Initializes available voices with polling fallback to catch asynchronous Edge Neural voices.
   */
  initVoices() {
    if (!this.synth || this.disposed) return;
    let pollCount = 0;
    const loadVoices = () => {
      if (this.disposed) return false;
      const voices = this.synth.getVoices();
      if (!voices?.length) return false;
      const previous = this.selectedVoice;
      this.voices = voices;
      const preferred = this.preferredVoiceId || this.getVoiceIdentity(previous);
      const matched = voices.find(v => this.getVoiceIdentity(v) === preferred || v.name === preferred);
      if (matched) this.selectedVoice = matched;
      else if (!previous) this.selectBestVietnameseVoice();
      // Preserve an explicit selection while the native voice list is incomplete.
      if (this.engine === 'browser' && this.getVoiceIdentity(previous) !== this.getVoiceIdentity(this.selectedVoice)) {
        this.scheduleConfigurationRestart();
      }
      if (this.voicePollTimer) {
        clearInterval(this.voicePollTimer);
        this.voicePollTimer = null;
      }
      this.emitPlaybackCallback(this.onVoicesLoaded, this.voices, this.selectedVoice);
      return true;
    };
    if (!loadVoices()) {
      this.voicePollTimer = setInterval(() => {
        if (loadVoices() || ++pollCount >= 12) {
          clearInterval(this.voicePollTimer);
          this.voicePollTimer = null;
        }
      }, 250);
      this.voicePollTimer?.unref?.();
    }
    this.voicesChangedHandler = loadVoices;
    this.synth.onvoiceschanged = this.voicesChangedHandler;
  }


  /**
   * Detects and picks the best Vietnamese voice available on the device.
   * Priority: Edge Online Natural (HoaiMy, Nam) -> Google tiếng Việt -> OS Vietnamese -> Default
   */
  selectBestVietnameseVoice() {
    if (!this.voices || this.voices.length === 0) return null;

    const viVoices = this.voices.filter(v =>
      v.lang && (v.lang.toLowerCase() === 'vi-vn' || v.lang.toLowerCase().startsWith('vi'))
    );

    if (viVoices.length > 0) {
      // Prioritize natural/online neural voices if present
      const neuralVoice = viVoices.find(v =>
        /natural|hoaimy|namminh|microsoft nam/i.test(v.name)
      );
      const googleVoice = viVoices.find(v =>
        /google/i.test(v.name)
      );
      this.selectedVoice = neuralVoice || googleVoice || viVoices[0];
    } else {
      this.selectedVoice = this.voices.find(v => v.default) || this.voices[0];
    }

    return this.selectedVoice;
  }

  /**
   * Switches TTS Engine: 'browser' or 'cloud'
   */
  setEngine(engine) {
    if (!['browser', 'cloud'].includes(engine) || this.engine === engine) return;
    this.engine = engine;
    this.scheduleConfigurationRestart();
  }


  /**
   * Updates Cloud BYOK configurations.
   */
  setCloudConfig(config) {
    const next = { ...this.cloudConfig, ...config };
    const changed = Object.keys(next).some(key => next[key] !== this.cloudConfig[key]);
    this.cloudConfig = next;
    if (changed && this.engine === 'cloud') this.scheduleConfigurationRestart();
  }


  /**
   * Retrieves lightweight observability metrics from the Prefetch Queue Scheduler.
   */
  getPrefetchMetrics() {
    return this.queueScheduler ? this.queueScheduler.getMetrics() : null;
  }

  /**
   * Enables or disables debug console logging for Cloud TTS prefetch.
   */
  setPrefetchDebugMode(enabled) {
    if (this.queueScheduler) {
      this.queueScheduler.setDebugMode(enabled);
    }
  }

  /**
   * Records a bounded diagnostics event into aggregate counters and bounded ring buffer.
   */
  recordDiagnosticsEvent(type, payload = {}) {
    if (this.diagnosticsCounts && this.diagnosticsCounts[type] !== undefined) {
      this.diagnosticsCounts[type]++;
    }
    if (!this.diagnosticsEnabled) return;
    const entry = {
      type,
      timestamp: Date.now(),
      generation: this.generationToken,
      chunkIndex: this.currentIndex,
      state: this.isPlaying ? (this.isPaused ? 'paused' : 'playing') : 'idle',
      origin: payload.origin || 'tts',
      ...payload
    };
    this.diagnosticsRingBuffer.push(entry);
    if (this.diagnosticsRingBuffer.length > this.maxDiagnosticsEvents) {
      this.diagnosticsRingBuffer.shift();
    }
    if (this.debugMode) {
      console.log(`[TTS-DIAG] ${entry.type} (chunk ${entry.chunkIndex}, gen ${entry.generation}, origin: ${entry.origin})`);
    }
  }

  /**
   * Returns a clean, bounded snapshot of playback lifecycle state and diagnostics counters.
   */
  getDiagnosticsSnapshot() {
    return {
      enabled: this.diagnosticsEnabled,
      debugMode: this.debugMode,
      generationId: this.generationToken,
      lastPlaybackOutcome: this.lastPlaybackOutcome ? { ...this.lastPlaybackOutcome } : null,
      hasConfigurationRestartTimer: this.configurationRestartTimer !== null,
      hasVoicePollTimer: this.voicePollTimer !== null,
      utteranceToken: this.utteranceToken,
      advanceSequenceToken: this.advanceSequenceToken,
      currentIndex: this.currentIndex,
      isPlaying: this.isPlaying,
      isPaused: this.isPaused,
      isAdvancingChunk: this.isAdvancingChunk,
      counts: { ...this.diagnosticsCounts },
      hasActiveUtterance: Boolean(this.currentUtterance),
      hasKeepAliveTimer: Boolean(this.keepAliveTimer),
      hasTransitionTimer: Boolean(this.transitionTimer),
      hasStuckWatchdogTimer: Boolean(this.stuckWatchdogTimer),
      hasCurrentAudioUrl: Boolean(this.currentAudioUrl),
      activeCloudRequestId: this.activeCloudRequestId,
      pendingCloudRequestId: this.pendingCloudRequestId,
      activeAudioResources: Boolean(this.currentAudioUrl || (this.audioElement && this.audioElement.src)),
      activeCloudRequests: this.queueScheduler ? (this.queueScheduler.activeFetches?.size || 0) : 0,
      activeTimers: {
        keepAlive: Boolean(this.keepAliveTimer),
        transition: Boolean(this.transitionTimer),
        watchdog: Boolean(this.stuckWatchdogTimer),
        configurationRestart: this.configurationRestartTimer !== null
      },
      recentEvents: [...this.diagnosticsRingBuffer]
    };
  }

  /**
   * Resets diagnostics counters and clears the event ring buffer.
   */
  resetDiagnostics() {
    this.diagnosticsCounts = {
      speak: 0,
      cancel: 0,
      pause: 0,
      resume: 0,
      onstart: 0,
      onend: 0,
      onerror: 0,
      autoAdvance: 0,
      stuck_watchdog: 0,
      keepalive_pulse: 0,
      keepalive_start: 0,
      keepalive_stop: 0,
      speakCurrentChunk: 0,
      speakBrowserChunk: 0,
      chapterComplete: 0
    };
    this.diagnosticsRingBuffer = [];
  }

  /**
   * Enables or disables diagnostics ring-buffer recording.
   */
  setDiagnosticsEnabled(enabled) {
    this.diagnosticsEnabled = Boolean(enabled);
    if (!this.diagnosticsEnabled) {
      this.diagnosticsRingBuffer = [];
    }
  }

  /**
   * Enables or disables debug console logging. Default is false.
   */
  setDebugMode(enabled) {
    this.debugMode = Boolean(enabled);
  }

  /**
   * Sets browser voice by URI or Name.
   */
  getVoiceIdentity(voice) {
    return voice ? (voice.voiceURI || [voice.name || '', voice.lang || ''].join('|')) : '';
  }

  getPlaybackConfigurationKey() {
    return JSON.stringify(this.engine === 'browser'
      ? ['browser', this.getVoiceIdentity(this.selectedVoice), this.rate, this.pitch]
      : ['cloud', this.cloudConfig, this.rate]);
  }

  clearConfigurationRestart() {
    if (this.configurationRestartTimer !== null) clearTimeout(this.configurationRestartTimer);
    this.configurationRestartTimer = null;
  }

  scheduleConfigurationRestart() {
    this.clearConfigurationRestart();
    if (!this.isPlaying || this.disposed) return;
    if (this.isPaused) {
      this.pausedChunkChanged = !this.browserPlayback ||
        this.browserPlayback.configurationKey !== this.getPlaybackConfigurationKey();
      return;
    }
    if (this.isAdvancingChunk) return; // Next chunk already picks up the new configuration.
    if (this.browserPlayback?.configurationKey === this.getPlaybackConfigurationKey()) return;
    const generation = this.generationToken;
    const request = this.playbackRequestId;
    const index = this.currentIndex;
    const timer = setTimeout(() => {
      if (this.configurationRestartTimer !== timer) return;
      this.configurationRestartTimer = null;
      if (this.disposed || !this.isPlaying || this.isPaused ||
          generation !== this.generationToken || request !== this.playbackRequestId ||
          index !== this.currentIndex) return;
      this.cancelCurrentSource();
      this.speakCurrentChunk();
    }, 250);
    this.configurationRestartTimer = timer;
    timer?.unref?.();
  }

  setVoice(voiceURI) {
    const found = (voiceURI && typeof voiceURI === 'object')
      ? voiceURI
      : this.voices.find(v => v.voiceURI === voiceURI || v.name === voiceURI);
    if (!found) {
      if (typeof voiceURI === 'string' && voiceURI) this.preferredVoiceId = voiceURI;
      return;
    }
    const changed = this.getVoiceIdentity(found) !== this.getVoiceIdentity(this.selectedVoice);
    this.preferredVoiceId = this.getVoiceIdentity(found);
    this.selectedVoice = found;
    if (changed && this.engine === 'browser') this.scheduleConfigurationRestart();
  }


  /**
   * Updates playback rate (0.7 - 2.2).
   */
  setRate(newRate) {
    const rate = Math.max(0.7, Math.min(2.2, parseFloat(newRate) || 1.0));
    if (rate === this.rate) return;
    this.rate = rate;
    this.scheduleConfigurationRestart();
  }


  /**
   * Updates voice pitch (0.5 - 1.5).
   */
  setPitch(newPitch) {
    const pitch = Math.max(0.5, Math.min(1.5, parseFloat(newPitch) || 1.0));
    if (pitch === this.pitch) return;
    this.pitch = pitch;
    if (this.engine === 'browser') this.scheduleConfigurationRestart();
  }


  /**
   * Updates playback volume (0.0 - 1.0) for sleep timer fade-out.
   */
  setVolume(newVolume) {
    this.volume = Math.max(0, Math.min(1, parseFloat(newVolume) !== undefined && !isNaN(parseFloat(newVolume)) ? parseFloat(newVolume) : 1.0));
    if (this.audioElement) {
      this.audioElement.volume = this.volume;
    }
  }

  /**
   * Toggles auto scrolling on/off.
   */
  setAutoScroll(enabled) {
    this.autoScrollEnabled = !!enabled;
  }

  /**
   * Loads raw text and slices it into TTS chunk sentences.
   */
  loadText(fullText) {
    this.stop();
    this.chunks = this.splitIntoSmartChunks(fullText);
    this.currentIndex = 0;
    if (this.queueScheduler) {
      this.queueScheduler.cancelAll();
    }
  }

  /**
   * Breaks text down into natural sentences and clauses suitable for TTS.
   */
  splitIntoSmartChunks(text) {
    if (!text || !text.trim()) return [];

    const rawParagraphs = text.split(/\n+/);
    const resultChunks = [];

    for (const paragraph of rawParagraphs) {
      const cleanPara = paragraph.trim();
      if (!cleanPara) continue;

      const sentenceRegex = /[^.!?…;:\n]+[.!?…;:\"'”’»\)\]]*|\S+/g;
      const rawSentences = cleanPara.match(sentenceRegex) || [cleanPara];

      for (let s of rawSentences) {
        s = s.trim();
        if (!s) continue;

        // Bỏ qua hoặc gộp chunk chỉ chứa ký tự đặc biệt/dấu câu không có âm vị phát âm
        if (!/[\p{L}\p{N}]/u.test(s)) {
          if (resultChunks.length > 0) {
            resultChunks[resultChunks.length - 1] += ' ' + s;
          }
          continue;
        }

        if (s.length > 220) {
          const subClauses = s.split(/,\s+/);
          let tempBuf = '';
          for (const sub of subClauses) {
            if ((tempBuf + sub).length < 180) {
              tempBuf += (tempBuf ? ', ' : '') + sub;
            } else {
              if (tempBuf && /[\p{L}\p{N}]/u.test(tempBuf)) resultChunks.push(tempBuf.trim());
              tempBuf = sub;
            }
          }
          if (tempBuf && /[\p{L}\p{N}]/u.test(tempBuf)) resultChunks.push(tempBuf.trim());
        } else {
          resultChunks.push(s);
        }
      }
    }

    return resultChunks;
  }

  /**
   * Starts playing from the current index.
   */
  play() {
    if (this.disposed || this.chunks.length === 0 || (this.isPlaying && !this.isPaused)) return;

    if (this.isPaused) {
      this.resume();
      return;
    }

    this.generationToken++;
    this.advanceSequenceToken++;
    this.isAdvancingChunk = false;
    this.autoChunkTransitionTimestamps = [];
    this.consecutiveErrorCount = 0;
    this.isPlaying = true;
    this.isPaused = false;
    const generation = this.generationToken;
    const request = this.playbackRequestId;
    this.emitStatus('playing');
    if (generation !== this.generationToken || request !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
    this.recordDiagnosticsEvent('play', { chunkIndex: this.currentIndex });

    if (this.synth?.paused) {
      try { this.synth.resume(); } catch (e) { /* ignore */ }
    }

    this.startKeepAlive();
    this.speakCurrentChunk();
  }

  /**
   * Pauses TTS playback across browser synth or cloud audio.
   */
  pause() {
    if (!this.isPlaying || this.isPaused) return;
    // Retained native utterances keep their generation. Only pending handoffs are invalidated.
    this.advanceSequenceToken++;
    this.isAdvancingChunk = false;
    if (this.configurationRestartTimer !== null) this.pausedChunkChanged = true;
    this.clearConfigurationRestart();
    this.isPaused = true;
    if (this.browserPlayback) this.browserPlayback.pausedAt = Date.now();
    if (this.transitionTimer) clearTimeout(this.transitionTimer);
    this.transitionTimer = null;
    this.clearStuckWatchdog();
    this.stopKeepAlive();
    this.recordDiagnosticsEvent('pause', { chunkIndex: this.currentIndex });
    if (this.engine === 'cloud' && this.pendingCloudRequestId === this.playbackRequestId) {
      this.playbackRequestId++;
      this.pendingCloudRequestId = null;
      this.activeCloudRequestId = null;
      this.pausedChunkChanged = true;
    }
    if (this.engine === 'cloud' && this.audioElement && !this.audioElement.paused) {
      this.audioElement.pause();
    } else if (this.synth) {
      try { this.synth.pause(); } catch (e) { /* ignore */ }
    }
    this.emitActualPlayback(false);
    this.emitStatus('paused');
  }


  /**
   * Resumes TTS playback.
   */
  resume() {
    if (!this.isPlaying || !this.isPaused || this.disposed) return;
    if (this.previewUtterance) {
      this.detachUtterance(this.previewUtterance);
      this.previewUtterance = null;
    }
    this.cleanupPreviewAudio();
    this.advanceSequenceToken++;
    this.isAdvancingChunk = false;
    this.recordDiagnosticsEvent('resume', { chunkIndex: this.currentIndex });
    const speakSelectedChunk = this.pausedChunkChanged || (this.engine === 'browser' && !this.currentUtterance);
    this.pausedChunkChanged = false;
    this.isPaused = false;
    const context = this.browserPlayback;
    if (context?.pausedAt !== null && context?.pausedAt !== undefined) {
      context.pausedDuration += Date.now() - context.pausedAt;
      context.pausedAt = null;
    }
    const generation = this.generationToken;
    const requestId = this.playbackRequestId;
    this.emitStatus('playing');
    if (generation !== this.generationToken || requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
    if (speakSelectedChunk) {
      this.cancelCurrentSource();
      this.speakCurrentChunk();
      return;
    }
    if (context && this.currentUtterance === context.utterance) {
      this.scheduleStuckWatchdog(context.utterance, context.token, context.generation, context.utterance.text.length);
      this.startKeepAlive();
    }
    if (this.synth?.paused) {
      try { this.synth.resume(); } catch (e) { /* ignore */ }
    }
    if (this.engine === 'cloud' && this.audioElement?.paused) {
      const request = this.playbackRequestId;
      this.audioElement.play().catch(err => {
        if (request !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
        console.warn('Lỗi tiếp tục phát cloud audio:', err);
        this.speakCurrentChunk();
      });
    }
  }


  /**
   * Stops TTS playback completely.
   */
  stop() {
    if (this._isStopping) return;
    this._isStopping = true;
    try {
      this.stopKeepAlive();
      this.clearStuckWatchdog();
      this.utteranceStartedTimestamp = null;
      this.generationToken++;
      this.advanceSequenceToken++;
      this.isAdvancingChunk = false;
      this.autoChunkTransitionTimestamps = [];
      this.recordDiagnosticsEvent('stop', { chunkIndex: this.currentIndex });
      if (this.transitionTimer) {
        clearTimeout(this.transitionTimer);
        this.transitionTimer = null;
      }
      this.emitActualPlayback(false);
      this.cancelCurrentSource();
      this.pausedChunkChanged = false;

      const wasActive = this.isPlaying || this.isPaused;
      this.isPlaying = false;
      this.isPaused = false;
      this.consecutiveErrorCount = 0;
      if (wasActive && typeof this.onChunkEnd === 'function') {
        this.onChunkEnd(this.currentIndex);
      }
      this.emitStatus('stopped');
    } finally {
      this._isStopping = false;
    }
  }

  dispose() {
    if (this.disposed) return;
    this.disposed = true;
    this.stop();
    if (this.voicePollTimer) clearInterval(this.voicePollTimer);
    this.voicePollTimer = null;
    if (this.synth?.onvoiceschanged === this.voicesChangedHandler) this.synth.onvoiceschanged = null;
    this.voicesChangedHandler = null;
    this.onVoicesLoaded = null;
    for (const { type, handler } of this.audioEventListeners) {
      this.audioElement?.removeEventListener?.(type, handler);
    }
    this.audioEventListeners = [];
  }

  cancelCurrentSource() {
    this.clearConfigurationRestart();
    this.stopKeepAlive();
    this.browserPlayback = null;
    this.playbackRequestId = (this.playbackRequestId || 0) + 1;
    this.generationToken++;
    this.advanceSequenceToken++;
    this.isAdvancingChunk = false;
    this.recordDiagnosticsEvent('cancel', { chunkIndex: this.currentIndex });
    if (this.transitionTimer) {
      clearTimeout(this.transitionTimer);
      this.transitionTimer = null;
    }
    this.clearStuckWatchdog();
    this.utteranceStartedTimestamp = null;
    this.emitActualPlayback(false);
    if (this.previewUtterance) {
      this.previewUtterance.onend = null;
      this.previewUtterance.onerror = null;
      this.previewUtterance = null;
    }
    this.cleanupPreviewAudio();
    this.pendingCloudRequestId = null;
    this.activeCloudRequestId = null;
    this.cloudChunkStartedIndex = null;
    this.lastChunkEndedTimestamp = null;
    if (this.queueScheduler) {
      this.queueScheduler.invalidate(this.generationToken);
    }

    if (this.currentUtterance) {
      this.currentUtterance.onend = null;
      this.currentUtterance.onerror = null;
      this.currentUtterance.onstart = null;
      this.currentUtterance.onpause = null;
      this.currentUtterance.onresume = null;
      this.currentUtterance = null;
    }

    if (this.synth) {
      try {
        this.synth.cancel();
      } catch (e) {
        // ignore
      }
      if (this.synth.paused) {
        try { this.synth.resume(); } catch (e) { /* ignore */ }
      }
    }

    if (this.audioElement && (this.currentAudioUrl || this.audioElement.src)) {
      if (typeof this.audioElement.pause === 'function') {
        this.audioElement.pause();
      }
      this.audioElement.currentTime = 0;
      this.cleanupCurrentAudioUrl();
    }
  }

  /**
   * Navigates to next sentence.
   */
  next() {
    if (this.currentIndex < this.chunks.length - 1) {
      this.generationToken++;
      this.advanceSequenceToken++;
      this.isAdvancingChunk = false;
      this.autoChunkTransitionTimestamps = [];
      this.consecutiveErrorCount = 0;
      if (this.transitionTimer) {
        clearTimeout(this.transitionTimer);
        this.transitionTimer = null;
      }
      this.currentIndex++;
      if (this.isPlaying && this.isPaused) {
        this.cancelCurrentSource();
        this.pausedChunkChanged = true;
        this.emitChunkStart();
      } else if (this.isPlaying) {
        this.speakCurrentChunk();
      } else {
        this.emitChunkStart();
      }
    }
  }

  /**
   * Navigates to previous sentence.
   */
  prev() {
    if (this.currentIndex > 0) {
      this.generationToken++;
      this.advanceSequenceToken++;
      this.isAdvancingChunk = false;
      this.autoChunkTransitionTimestamps = [];
      this.consecutiveErrorCount = 0;
      if (this.transitionTimer) {
        clearTimeout(this.transitionTimer);
        this.transitionTimer = null;
      }
      this.currentIndex--;
      if (this.isPlaying && this.isPaused) {
        this.cancelCurrentSource();
        this.pausedChunkChanged = true;
        this.emitChunkStart();
      } else if (this.isPlaying) {
        this.speakCurrentChunk();
      } else {
        this.emitChunkStart();
      }
    }
  }

  /**
   * Jumps directly to a given sentence chunk.
   */
  jumpToIndex(index, autoPlay = true) {
    if (this.disposed) return;
    if (index >= 0 && index < this.chunks.length) {
      this.generationToken++;
      this.advanceSequenceToken++;
      this.isAdvancingChunk = false;
      this.autoChunkTransitionTimestamps = [];
      this.consecutiveErrorCount = 0;
      if (this.transitionTimer) {
        clearTimeout(this.transitionTimer);
        this.transitionTimer = null;
      }
      this.currentIndex = index;
      if (this.queueScheduler) {
        this.queueScheduler.invalidate(this.generationToken);
      }
      if (autoPlay) {
        this.pausedChunkChanged = false;
        this.isPlaying = true;
        this.isPaused = false;
        const generation = this.generationToken;
        const request = this.playbackRequestId;
        this.emitStatus('playing');
        if (generation !== this.generationToken || request !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
        this.startKeepAlive();
        this.speakCurrentChunk();
      } else {
        if (this.isPlaying && this.isPaused) {
          this.cancelCurrentSource();
          this.pausedChunkChanged = true;
        }
        this.emitChunkStart();
      }
    }
  }

  /**
   * Speaks the current chunk based on active engine.
   */
  speakCurrentChunk() {
    if (this.disposed || !this.isPlaying || this.isPaused) return;
    this.clearConfigurationRestart();
    if (!this.chunks || this.chunks.length === 0) {
      this.stop();
      return;
    }

    if (this.currentIndex >= this.chunks.length) {
      this.stop();
      this.emitStatus('completed');
      return;
    }

    const chunkText = this.chunks[this.currentIndex];
    const requestId = this.playbackRequestId = (this.playbackRequestId || 0) + 1;
    this.recordDiagnosticsEvent('speakCurrentChunk', { chunkIndex: this.currentIndex, origin: 'speakCurrentChunk' });
    this.pendingCloudRequestId = null;
    this.activeCloudRequestId = null;
    this.cloudChunkStartedIndex = null;

    // Chỉ tạm dừng và giải phóng audio element nếu thực sự có URL hoặc audio source đang nạp (Cloud TTS)
    if (this.currentAudioUrl || (this.audioElement && this.audioElement.src)) {
      if (this.audioElement && typeof this.audioElement.pause === 'function') {
        this.audioElement.pause();
      }
      if (this.audioElement) {
        this.audioElement.currentTime = 0;
      }
      this.cleanupCurrentAudioUrl();
    }

    if (this.engine === 'cloud' && this.cloudConfig.apiKey) {
      return this.speakCloudChunk(chunkText, requestId);
    } else {
      return this.speakBrowserChunk(chunkText, requestId);
    }
  }

  /**
   * Tier 1: Browser Web Speech API playback with pitch & rate tuning.
   * Dọn dẹp Utterance an toàn tuyệt đối, chống rò rỉ bộ nhớ và chặn lặp vô hạn Chromium.
   */
  detachUtterance(utterance) {
    if (!utterance) return;
    utterance.onstart = utterance.onend = utterance.onerror = null;
    utterance.onpause = utterance.onresume = null;
  }

  // Lifecycle anomaly checks cannot determine whether native audio was audible.
  silentCompletionGuard(context) {
    if (context.startedAt === null) return 'missing-start';
    const letters = (context.utterance.text.match(/[\p{L}\p{N}]/gu) || []).length;
    if (letters < 80) return null;
    const activeMs = Math.max(0, Date.now() - context.startedAt - context.pausedDuration);
    return activeMs < Math.min(200, letters * 2 / context.utterance.rate) ? 'premature-end' : null;
  }

  haltBrowserPlayback(reason, outcome = 'error') {
    const index = this.currentIndex;
    this.lastPlaybackOutcome = { type: outcome, reason, index };
    this.isPlaying = this.isPaused = false;
    this.cancelCurrentSource();
    this.pausedChunkChanged = false;
    const generation = this.generationToken;
    this.emitPlaybackCallback(this.onChunkEnd, index);
    if (this.generationToken !== generation || this.isPlaying) return;
    this.emitStatus('error');
    if (this.generationToken !== generation || this.isPlaying) return;
    this.emitPlaybackCallback(this.onFallbackWarning,
      'Giọng đọc gặp lỗi hoặc phản hồi bất thường (' + reason + '). Đã dừng tại câu hiện tại; hãy bấm Phát để thử lại.');
  }

  speakBrowserChunk(chunkText, requestId = this.playbackRequestId) {
    if (!this.synth || this.disposed || requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
    const activeGeneration = this.generationToken;
    this.recordDiagnosticsEvent('speakBrowserChunk', { chunkIndex: this.currentIndex, origin: 'speakBrowserChunk' });
    if (this.currentUtterance) {
      this.detachUtterance(this.currentUtterance);
      this.currentUtterance = this.browserPlayback = null;
      this.recordDiagnosticsEvent('cancel', { chunkIndex: this.currentIndex, origin: 'speakBrowserChunk:cleanup' });
      try { this.synth.cancel(); } catch (e) { /* ignore */ }
    }
    this.clearStuckWatchdog();
    this.stopKeepAlive();
    if (this.synth.paused) {
      this.recordDiagnosticsEvent('resume', { chunkIndex: this.currentIndex, origin: 'speakBrowserChunk:unpause' });
      try { this.synth.resume(); } catch (e) { /* ignore */ }
    }
    let validText = String(chunkText ?? '');
    if (!validText.trim() || !/[\p{L}\p{N}]/u.test(validText)) {
      while (this.currentIndex < this.chunks.length) {
        const text = String(this.chunks[this.currentIndex] ?? '');
        if (text.trim() && /[\p{L}\p{N}]/u.test(text)) break;
        this.currentIndex++;
      }
      if (this.currentIndex >= this.chunks.length) {
        this.stop();
        this.emitStatus('completed');
        return;
      }
      validText = String(this.chunks[this.currentIndex]);
    }
    const utterance = new SpeechSynthesisUtterance(validText);
    utterance.voice = this.selectedVoice;
    utterance.lang = this.selectedVoice?.lang || 'vi-VN';
    utterance.rate = this.rate; utterance.pitch = this.pitch; utterance.volume = this.volume;
    const context = {
      utterance, generation: activeGeneration, token: ++this.utteranceToken,
      requestId, index: this.currentIndex, terminalHandled: false,
      startedAt: null, pausedAt: null, pausedDuration: 0,
      configurationKey: this.getPlaybackConfigurationKey()
    };
    this.currentUtterance = utterance; this.browserPlayback = context;
    const sameSession = () => this.generationToken === context.generation &&
      this.utteranceToken === context.token && this.playbackRequestId === context.requestId &&
      this.currentIndex === context.index && this.isPlaying && !this.disposed;
    const ownsPlayback = () => sameSession() && this.currentUtterance === utterance && this.browserPlayback === context;
    const finishOwned = () => {
      context.terminalHandled = true;
      this.detachUtterance(utterance);
      this.clearStuckWatchdog(); this.stopKeepAlive(); this.clearConfigurationRestart();
      this.utteranceStartedTimestamp = null;
      this.currentUtterance = this.browserPlayback = null;
    };
    utterance.onstart = () => {
      if (context.terminalHandled || !ownsPlayback() || this.isPaused || context.startedAt !== null) return;
      context.startedAt = Date.now(); this.utteranceStartedTimestamp = context.startedAt;
      this.scheduleStuckWatchdog(utterance, context.token, context.generation, validText.length);
      this.recordDiagnosticsEvent('onstart', { chunkIndex: context.index });
      this.emitPlaybackCallback(this.onActualChunkStart, context.index);
      if (!ownsPlayback() || this.isPaused) return;
      this.emitActualPlayback(true);
      if (ownsPlayback() && !this.isPaused) this.emitChunkStart();
    };
    utterance.onpause = () => { if (ownsPlayback() && !context.terminalHandled) this.emitActualPlayback(false); };
    utterance.onresume = () => { if (ownsPlayback() && !context.terminalHandled && !this.isPaused) this.emitActualPlayback(true); };
    utterance.onend = () => {
      if (context.terminalHandled || !ownsPlayback()) return;
      const paused = this.isPaused;
      const invalidCompletion = this.silentCompletionGuard(context);
      finishOwned();
      if (paused) { this.pausedChunkChanged = true; return; }
      if (invalidCompletion) { this.haltBrowserPlayback(invalidCompletion, 'suspicious-completion'); return; }
      const completedIndex = context.index;
      this.lastPlaybackOutcome = { type: 'normal-completion', index: completedIndex };
      this.recordDiagnosticsEvent('onend', { chunkIndex: completedIndex });
      this.consecutiveErrorCount = 0;
      if (completedIndex + 1 >= this.chunks.length) this.emitActualPlayback(false);
      if (!sameSession() || this.isPaused) return;
      this.emitPlaybackCallback(this.onActualChunkComplete, completedIndex);
      if (!sameSession() || this.isPaused) return;
      this.emitPlaybackCallback(this.onChunkEnd, completedIndex);
      if (!sameSession() || this.isPaused || this.checkChunkVelocityCircuitBreaker()) return;
      if (completedIndex + 1 < this.chunks.length) {
        const advanceToken = ++this.advanceSequenceToken;
        this.isAdvancingChunk = true;
        this.recordDiagnosticsEvent('autoAdvance', { nextIndex: completedIndex + 1 });
        const timer = setTimeout(() => {
          if (this.transitionTimer !== timer) return;
          this.transitionTimer = null; this.isAdvancingChunk = false;
          if (!sameSession() || this.isPaused || advanceToken !== this.advanceSequenceToken) return;
          this.currentIndex = completedIndex + 1; this.speakCurrentChunk();
        }, 0);
        this.transitionTimer = timer;
      } else {
        this.recordDiagnosticsEvent('chapterComplete', { chunkIndex: completedIndex, origin: 'onNaturalComplete' });
        this.emitPlaybackCallback(this.onNaturalComplete, completedIndex);
        if (!sameSession() || this.isPaused) return;
        this.stop();
        if (!this.isPlaying && this.playbackRequestId === requestId + 1) this.emitStatus('completed');
      }
    };
    utterance.onerror = event => {
      if (context.terminalHandled || !ownsPlayback()) return;
      const reason = event?.error || 'unknown';
      finishOwned(); this.emitActualPlayback(false);
      if (!sameSession()) return;
      this.recordDiagnosticsEvent('onerror', { chunkIndex: context.index, error: reason });
      const outcome = reason === 'canceled' ? 'cancel' : reason === 'interrupted' ? 'interrupted' : 'error';
      this.lastPlaybackOutcome = { type: outcome, reason, index: context.index };
      if (this.isPaused && (reason === 'canceled' || reason === 'interrupted')) {
        this.pausedChunkChanged = true; return;
      }
      this.consecutiveErrorCount++;
      if (reason === 'network' && this.consecutiveErrorCount <= 1 && !this.isPaused) {
        this.emitPlaybackCallback(this.onChunkEnd, context.index);
        if (!sameSession() || this.isPaused) return;
        const advanceToken = ++this.advanceSequenceToken;
        this.isAdvancingChunk = true;
        const timer = setTimeout(() => {
          if (this.transitionTimer !== timer) return;
          this.transitionTimer = null; this.isAdvancingChunk = false;
          if (!sameSession() || this.isPaused || advanceToken !== this.advanceSequenceToken) return;
          this.speakCurrentChunk();
        }, 0);
        this.transitionTimer = timer;
      } else this.haltBrowserPlayback(reason, outcome);
    };
    this.scheduleStuckWatchdog(utterance, context.token, context.generation, validText.length);
    this.startKeepAlive();
    this.recordDiagnosticsEvent('speak', { chunkIndex: context.index, textLen: validText.length });
    try { this.synth.speak(utterance); }
    catch (error) { utterance.onerror?.({ error: 'synthesis-failed' }); }
  }


  /**
   * Tier 2: Cloud AI TTS playback with IndexedDB caching & automatic fallback.
   */
  async speakCloudChunk(chunkText, requestId = this.playbackRequestId) {
    this.pendingCloudRequestId = requestId;
    this.emitChunkStart();

    const provider = this.cloudConfig.provider || 'fpt';
    const voice = this.cloudConfig.voice || 'banmai';
    const model = this.cloudConfig.model || '';
    const cacheKey = buildCacheKey({
      text: chunkText,
      provider,
      voice,
      model,
      rate: this.rate
    });
    const metadata = { provider, voice, model, rate: this.rate, textHash: hashText(chunkText) };

    try {
      // 1. Request audio chunk via Queue Scheduler (cache-first, coalesced in-flight, or bounded fetch)
      let audioBlob = null;
      if (this.queueScheduler) {
        audioBlob = await this.queueScheduler.requestChunk({
          chunkIndex: this.currentIndex,
          text: chunkText,
          config: this.cloudConfig,
          rate: this.rate,
          generation: this.generationToken
        });
      } else {
        try {
          audioBlob = await getCachedAudio(cacheKey, metadata);
        } catch (cacheErr) {
          console.warn('Lỗi truy cập cache IndexedDB:', cacheErr);
        }
        if (requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;

        if (!audioBlob) {
          audioBlob = await this.fetchCloudAudio(chunkText, provider, voice);
          if (requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
          if (audioBlob) {
            setCachedAudio(cacheKey, audioBlob, provider, voice, metadata).catch(e => console.warn(e));
          }
        }
      }
      if (requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;

      if (!audioBlob) {
        throw new Error('Không nhận được dữ liệu âm thanh từ Cloud API.');
      }

      // 3. Play audio blob via Audio element
      this.cleanupCurrentAudioUrl();
      if (typeof URL !== 'undefined' && URL.createObjectURL && this.audioElement) {
        this.currentAudioUrl = URL.createObjectURL(audioBlob);
        this.activeCloudRequestId = requestId;
        this.cloudChunkStartedIndex = null;
        this.audioElement.src = this.currentAudioUrl;
        this.audioElement.playbackRate = 1.0; // Rate is controlled at generation or natively
        this.audioElement.volume = this.volume;
        await this.audioElement.play();
        if (this.lastChunkEndedTimestamp) {
          const gapMs = Math.max(0, Date.now() - this.lastChunkEndedTimestamp);
          if (this.queueScheduler) {
            this.queueScheduler.recordChunkGap(gapMs);
          }
          this.lastChunkEndedTimestamp = null;
        }
        if (this.queueScheduler && this.chunks && this.chunks.length > 0 && this.cloudChunkStartedIndex !== this.currentIndex) {
          this.cloudChunkStartedIndex = this.currentIndex;
          this.emitPlaybackCallback(this.onActualChunkStart, this.currentIndex);
          this.queueScheduler.markPlaying(this.currentIndex);
          this.queueScheduler.updateWindow({
            currentIndex: this.currentIndex,
            chunks: this.chunks,
            config: this.cloudConfig,
            rate: this.rate,
            generation: this.generationToken
          });
        }
      } else {
        // Fallback for environment without HTMLAudioElement
        if (requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
        this.speakBrowserChunk(chunkText, requestId);
      }

    } catch (err) {
      if (requestId !== this.playbackRequestId || !this.isPlaying || this.isPaused) return;
      const isAbort = err?.name === 'AbortError' || /abort/i.test(err?.message || '');
      if (isAbort) return;

      const shouldFallback = this.activeCloudRequestId === requestId || this.pendingCloudRequestId === requestId;
      this.activeCloudRequestId = null;
      this.cleanupCurrentAudioUrl();
      console.warn(`Lỗi Cloud TTS [${provider}]:`, err.message);
      if (shouldFallback) {
        if (this.queueScheduler) {
          this.queueScheduler.invalidate(this.generationToken);
        }
        if (typeof this.onFallbackWarning === 'function') {
          this.onFallbackWarning(`Cloud AI (${provider}) lỗi: ${err.message}. Đã tự động chuyển về giọng trình duyệt.`);
        }
        // Graceful fallback to Browser Web Speech API
        this.engine = 'browser';
        this.speakBrowserChunk(chunkText, requestId);
      }
    } finally {
      if (this.pendingCloudRequestId === requestId) this.pendingCloudRequestId = null;
    }
  }

  /**
   * Fetches audio blob from Cloud AI provider (FPT AI, OpenAI, or Custom).
   */
  async fetchCloudAudio(text, provider, voice, signal = null) {
    return await executeFetchCloudAudio({
      text,
      provider: provider || this.cloudConfig.provider,
      voice: voice || this.cloudConfig.voice,
      config: this.cloudConfig,
      rate: this.rate,
      signal
    });
  }

  /**
   * Polls and downloads audio file from FPT CDN until ready.
   */
  async downloadAudioWithRetry(url, maxRetries = 6, delayMs = 800, signal = null) {
    return await executeDownloadAudioWithRetry(url, maxRetries, delayMs, signal);
  }

  /**
   * Previews a voice by speaking a short Vietnamese sentence:
   * "Chào bạn, tôi là giọng đọc tiếng Việt của VietReader."
   */
  async previewVoice(voiceIdOrUri, engine = 'browser', pitch = 1.0, rate = 1.0) {
    if (this.disposed) return false;
    const sampleText = 'Chào bạn, tôi là giọng đọc tiếng Việt của VietReader.';

    const wasPlaying = this.isPlaying && !this.isPaused;
    if (engine === 'cloud' && this.cloudConfig.apiKey) {
      this.cleanupPreviewAudio();
      if (this.isPlaying) {
        this.cancelCurrentSource();
        this.pausedChunkChanged = true;
        this.isPaused = true;
        if (wasPlaying) this.emitStatus('paused');
      } else {
        this.cancelCurrentSource();
      }
      const previewRequestId = this.previewRequestId;
      try {
        const provider = this.cloudConfig.provider || 'fpt';
        const voice = voiceIdOrUri || this.cloudConfig.voice || 'banmai';
        const blob = await this.fetchCloudAudio(sampleText, provider, voice);
        if (previewRequestId !== this.previewRequestId) return false;
        const AudioCtor = typeof Audio !== 'undefined' ? Audio : globalThis.window?.Audio;
        if (blob && typeof URL !== 'undefined' && AudioCtor) {
          const url = URL.createObjectURL(blob);
          const previewAudio = new AudioCtor(url);
          this.previewAudio = previewAudio;
          this.previewAudioUrl = url;
          const cleanup = () => {
            if (this.previewAudio !== previewAudio) return;
            this.cleanupPreviewAudio();
            if (wasPlaying && this.isPlaying && this.isPaused) this.resume();
          };
          previewAudio.onended = cleanup;
          previewAudio.onerror = cleanup;
          try {
            await previewAudio.play();
            if (previewRequestId !== this.previewRequestId) return false;
          } catch (err) {
            if (this.previewAudio === previewAudio) {
              previewAudio.onended = null;
              previewAudio.onerror = null;
              try { previewAudio.pause(); } catch (pauseError) { /* ignore */ }
              this.previewAudio = null;
              if (this.previewAudioUrl === url) {
                try { URL.revokeObjectURL(url); } catch (revokeError) { /* ignore */ }
                this.previewAudioUrl = null;
              }
            }
            if (wasPlaying && this.isPlaying && this.isPaused) this.resume();
            throw err;
          }
          return true;
        }
      } catch (err) {
        if (wasPlaying && this.isPlaying && this.isPaused) this.resume();
        if (previewRequestId !== this.previewRequestId) return false;
        console.warn('Lỗi nghe thử Cloud Voice:', err);
        throw err;
      }
    }

    if (!this.synth) return false;

    if (this.isPlaying) {
      this.cancelCurrentSource();
      this.pausedChunkChanged = true;
      this.isPaused = true;
      if (wasPlaying) this.emitStatus('paused');
    } else {
      this.cancelCurrentSource();
    }

    const utterance = new SpeechSynthesisUtterance(sampleText);
    this.previewUtterance = utterance;
    utterance.lang = 'vi-VN';
    utterance.rate = rate;
    utterance.pitch = pitch;

    const previewGeneration = this.generationToken;
    const previewPlaybackRequest = this.playbackRequestId;
    const finishPreview = () => {
      if (this.previewUtterance !== utterance || this.generationToken !== previewGeneration ||
          this.playbackRequestId !== previewPlaybackRequest) return;
      utterance.onend = null;
      utterance.onerror = null;
      if (this.previewUtterance === utterance) this.previewUtterance = null;
      if (wasPlaying && this.isPlaying && this.isPaused) this.resume();
    };
    utterance.onend = finishPreview;
    utterance.onerror = finishPreview;

    const matchedVoice = this.voices.find(v => v.voiceURI === voiceIdOrUri || v.name === voiceIdOrUri);
    if (matchedVoice) utterance.voice = matchedVoice;
    else if (this.selectedVoice) utterance.voice = this.selectedVoice;

    this.synth.speak(utterance);
    return true;
  }
  cleanupPreviewAudio() {
    this.previewRequestId = (this.previewRequestId || 0) + 1;
    if (this.previewAudio) {
      this.previewAudio.onended = null;
      this.previewAudio.onerror = null;
      try { this.previewAudio.pause(); } catch (e) { /* ignore */ }
      this.previewAudio = null;
    }
    if (this.previewAudioUrl && typeof URL !== 'undefined' && URL.revokeObjectURL) {
      try { URL.revokeObjectURL(this.previewAudioUrl); } catch (e) { /* ignore */ }
      this.previewAudioUrl = null;
    }
  }

  cleanupCurrentAudioUrl() {
    if (this.audioElement && (this.currentAudioUrl || this.audioElement.src)) {
      try {
        this.audioElement.removeAttribute('src');
        if (typeof this.audioElement.load === 'function') {
          this.audioElement.load();
        }
      } catch (e) {
        // ignore
      }
    }
    if (this.currentAudioUrl && typeof URL !== 'undefined' && URL.revokeObjectURL) {
      try {
        URL.revokeObjectURL(this.currentAudioUrl);
      } catch (e) {
        // ignore
      }
      this.currentAudioUrl = null;
    }
  }

  emitChunkStart() {
    if (typeof this.onChunkStart === 'function') {
      this.onChunkStart(
        this.currentIndex,
        this.chunks[this.currentIndex] || '',
        this.chunks.length
      );
    }
  }

  emitStatus(status) {
    if (typeof this.onStatusChange === 'function') {
      this.onStatusChange(status);
    }
  }

  /**
   * Kiểm tra xem giọng đọc có phải là giọng trực tuyến / Natural Neural stream hay không.
   * Giọng trực tuyến truyền tải qua WebSocket trong Browser Process; tuyệt đối không pulse pause/resume.
   */
  isNaturalOrOnlineVoice(voice = this.selectedVoice) {
    if (!voice) return true; // Safe default: Không phát xung nếu chưa nhận diện được voice
    const name = voice.name || '';
    const uri = voice.voiceURI || '';
    return /natural|online|neural|hoaimy|namminh|microsoft nam|google/i.test(name) ||
           /natural|online|neural|hoaimy|namminh|microsoft nam|google/i.test(uri);
  }

  startKeepAlive() {
    this.stopKeepAlive();
    const context = this.browserPlayback;
    if (this.engine !== 'browser' || !this.isPlaying || this.isPaused || !context) return;
    this.recordDiagnosticsEvent('keepalive_start', { chunkIndex: this.currentIndex, origin: 'startKeepAlive' });
    const timer = setInterval(() => {
      if (this.keepAliveTimer !== timer) return;
      if (this.browserPlayback !== context || this.generationToken !== context.generation ||
          this.utteranceToken !== context.token || this.playbackRequestId !== context.requestId ||
          !this.isPlaying || this.isPaused || this.currentUtterance !== context.utterance) {
        this.stopKeepAlive(); return;
      }
      if (this.engine !== 'browser' || !this.synth?.speaking || this.synth.paused ||
          this.isAdvancingChunk || context.startedAt === null || this.configurationRestartTimer !== null) return;
      if (this.isNaturalOrOnlineVoice(context.utterance.voice)) return;
      const elapsed = Date.now() - context.startedAt - context.pausedDuration;
      if (elapsed < 10000) return;
      this.recordDiagnosticsEvent('keepalive_pulse', { chunkIndex: context.index, elapsed });
      try {
        this.synth.pause();
        if (this.keepAliveTimer === timer && this.browserPlayback === context &&
            this.generationToken === context.generation && this.isPlaying && !this.isPaused &&
            this.currentUtterance === context.utterance) this.synth.resume();
      } catch (e) { /* ignore */ }
    }, 10000);
    this.keepAliveTimer = timer;
    timer?.unref?.();
  }


  stopKeepAlive() {
    if (this.keepAliveTimer) {
      clearInterval(this.keepAliveTimer);
      this.keepAliveTimer = null;
      this.recordDiagnosticsEvent('keepalive_stop', { chunkIndex: this.currentIndex, origin: 'stopKeepAlive' });
    }
  }

  /**
   * Thiết lập watchdog giám sát native SpeechSynthesis.
   * Nếu trình duyệt Edge bị kẹt (treo không phát onend/onerror), dừng phát an toàn và chuyển sang trạng thái lỗi.
   * TUYỆT ĐỐI KHÔNG tự động emit onActualChunkComplete (hoàn tất ảo), KHÔNG tự tăng currentIndex, và KHÔNG tự động chuyển câu lặp vô hạn.
   */
  scheduleStuckWatchdog(utterance, currentUttToken, activeGeneration, textLength, customTimeoutMs = null) {
    if (this.disposed || this.currentUtterance !== utterance || this.utteranceToken !== currentUttToken ||
        this.generationToken !== activeGeneration || !this.isPlaying || this.isPaused) return;
    this.clearStuckWatchdog();
    const watchdogTimeoutMs = customTimeoutMs ?? Math.max(14000, (textLength || 50) * 200);
    const timer = setTimeout(() => {
      if (this.stuckWatchdogTimer !== timer) return;
      this.stuckWatchdogTimer = null;
      if (this.generationToken !== activeGeneration || this.utteranceToken !== currentUttToken ||
          !this.isPlaying || this.isPaused || this.currentUtterance !== utterance) return;
      this.recordDiagnosticsEvent('stuck_watchdog', { chunkIndex: this.currentIndex, timeoutMs: watchdogTimeoutMs });
      this.haltBrowserPlayback('watchdog-timeout', 'timeout');
    }, watchdogTimeoutMs);
    this.stuckWatchdogTimer = timer;
    timer?.unref?.();
  }


  clearStuckWatchdog() {
    if (this.stuckWatchdogTimer) {
      clearTimeout(this.stuckWatchdogTimer);
      this.stuckWatchdogTimer = null;
    }
  }
}
